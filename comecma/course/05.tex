\chapter{Récapitulatif général pour l'examen}
% =========================================================

\begin{rembox}[title=Ce chapitre]
Tout ce qu'il faut avoir en tête le jour de l'examen (sans formulaire !) : d'abord le cours
entier expliqué à une chèvre, puis la liste officielle des \textbf{formules à connaître par
c\oe ur} (chapitres~1 à~4 ; celles des polymères et composites s'ajouteront avec ces
chapitres), les méthodes pas à pas des calculs typiques et les pièges classiques.
\end{rembox}

\begin{goatbox}[title={Tout le cours à une chèvre}]
On apprend d'abord à \og tester\fg{} un matériau : le tirer, l'écraser, le rayer, le plier, le
laisser fluer, le casser. Puis on regarde trois familles. Les céramiques sont très rigides mais
cassent au moindre défaut, et ce défaut est le fruit du hasard : on parle de probabilité de
casser. Les métaux, eux, se déforment avant de casser grâce aux dislocations ; pour les rendre
plus solides, on sème des obstacles sur leur chemin (autres dislocations, atomes étrangers,
petits précipités, joints de grains, grains durs). L'acier ajoute un tour de magie : en le
refroidissant très vite, il se transforme d'un coup en martensite, très dure, qu'on adoucit un
peu ensuite pour qu'elle ne casse pas.
\end{goatbox}

\section{Formulaire complet (liste officielle du cours)}

\begin{keybox}[title=Chapitre 1 --- Les essais mécaniques]
\begin{itemize}
  \item $e = \dfrac{L - L_0}{L_0}$, \quad $\varepsilon = \displaystyle\int_{L_0}^{L}\frac{dL}{L} = \ln\frac{L}{L_0} = \ln(1+e)$
  \item $R = P/S_0$, \quad $\sigma = P/S \approx \dfrac{P}{S_0}\dfrac{L}{L_0} = R(1+e)$
  \item Considère : striction quand $\sigma = \dfrac{d\sigma}{d\varepsilon}$, ce qui correspond à $\dfrac{dR}{de} = 0$
  \item Flexion (Euler--Bernoulli) : $\varepsilon_{xx} = y/R$ ($y$ : distance à l'axe neutre, $R$ : rayon de courbure)
  \item $K = Y\sigma\sqrt{\pi a}$ ; rupture quand $K = K_c$, donc à $\sigma = \dfrac{K_c}{Y\sqrt{\pi a}}$
  \item À chaud (fluage) : $\dot\varepsilon = \dfrac{d\varepsilon}{dt} = A\sigma^N e^{-Q/RT}$ ;
        vitesse : $\sigma = K\dot\varepsilon^m$, $m = 1/N$
\end{itemize}
\end{keybox}

\begin{keybox}[title=Chapitre 2 --- Les céramiques]
\begin{itemize}
  \item Fluide newtonien : $\tau = \eta\dot\gamma = \eta\dfrac{d\gamma}{dt}$, ou en uniaxial
        $\sigma = 3\eta\dot\varepsilon$
  \item Contrainte thermique (2D, élastique, couche collée à un substrat indéformable) :
        $\sigma = \dfrac{E}{1-\nu}\,\alpha\,(T_1 - T_2)$
  \item Weibull : $S = 1 - F = \exp\!\Big[-\dfrac{V}{V_0}\Big(\dfrac{\sigma}{\sigma_0}\Big)^m\Big]$,
        soit $\ln S(\sigma,V) = \ln[1 - F(\sigma,V)] = -\dfrac{V}{V_0}\Big(\dfrac{\sigma}{\sigma_0}\Big)^m$,
        avec $\ln[1-F] \approx -F$ si $F \ll 1$
  \item Fissuration sous-critique : $\dfrac{da}{dt} = F(P_{\text{H}_2\text{O}}, K)\,e^{-Q/RT}$ avec
        $K = Y\sigma\sqrt{\pi a}$
\end{itemize}
\end{keybox}

\begin{keybox}[title={Chapitre 3 --- Métaux et alliages 1 : Al \& Cu}]
\begin{itemize}
  \item Loi du levier : $f_\alpha = \dfrac{C_\beta - C}{C_\beta - C_\alpha}$,
        $f_\beta = \dfrac{C - C_\alpha}{C_\beta - C_\alpha}$
  \item Maturation : $l = k\,t^{1/3}$ (forme complète $\lambda^3 - \lambda_0^3 = Kt$)
  \item Contournement de précipités non cisaillés : $\sigma = \dfrac{2Gb}{d} \approx \dfrac{Ea}{d}$
\end{itemize}
\end{keybox}

\begin{keybox}[title={Chapitre 4 --- Le fer et ses alliages}]
\begin{itemize}
  \item Hall--Petch : $\sigma(e = x\,\%) = \sigma_0 + k\,D^{-1/2}$
\end{itemize}
\end{keybox}

\begin{rembox}[title={À savoir aussi, même si ce n'est pas dans la liste}]
$G_c = AK_c^2/E$ ; $H \approx 3\sigma_y$ ; $\Delta P = 2\gamma/R$ ; indices $E/\rho$ et
$E/\rho^2$ ; compression des céramiques $\sigma = CK_c/(Y\sqrt{\pi\bar a})$, $C \approx 15$ ;
$(\sigma/\sigma_{TS})^n = t_{\text{test}}/t$ ; $F \le (1+i)^{-n}$ ; $M_s$ décroît avec C ;
$\eta = Ke^{Q/RT}$.
\end{rembox}

\section{Les méthodes pas à pas}

\begin{exbox}[title={Exploiter une courbe de traction}]
\begin{enumerate}
  \item Convertir : $\varepsilon = \ln(1+e)$, $\sigma = R(1+e)$ (valable tant que la déformation est
        homogène, donc jusqu'à $R_m$).
  \item $E$ = pente initiale ; $R_{0,2}$ = intersection avec la parallèle décalée de 0,2\,\%.
  \item Striction : au maximum de $R(e)$ ; ou, sur $\sigma(\varepsilon)$, là où
        $d\sigma/d\varepsilon = \sigma$ (pour $\sigma = K\varepsilon^n$ : $\varepsilon = n$).
  \item Énergie : aire sous la courbe ; la part élastique est l'aire sous la droite de décharge.
\end{enumerate}
\end{exbox}

\begin{exbox}[title={Problème de rupture}]
\begin{enumerate}
  \item Identifier la géométrie : fissure interne (longueur $2a$, $Y = 1$) ou de bord
        (longueur $a$, $Y = 1{,}12$).
  \item Rupture si $Y\sigma\sqrt{\pi a} \ge K_c$ : soit $\sigma_r = K_c/(Y\sqrt{\pi a})$, soit la
        taille critique $a_c = \frac{1}{\pi}\big(\frac{K_c}{Y\sigma}\big)^2$.
  \item Comparer à la limite d'élasticité : si $\sigma_r > \sigma_y$, la pièce plastifie avant de
        rompre (cas des métaux tenaces).
  \item Fatigue : $\Delta K = Y\Delta\sigma\sqrt{\pi a}$ et intégrer $da/dN = A(\Delta K)^m$ jusqu'à
        $a_c$.
\end{enumerate}
\end{exbox}

\begin{exbox}[title={Problème de Weibull}]
\begin{enumerate}
  \item Écrire $S = \exp[-k_m(V/V_0)(\sigma_{\max}/\sigma_0)^m]$ ($k_m = 1$ en traction).
  \item Deux pièces à même probabilité : $k_{m,1}V_1\sigma_1^m = k_{m,2}V_2\sigma_2^m$.
  \item Petite probabilité de rupture : $F \approx k_m(V/V_0)(\sigma/\sigma_0)^m$.
  \item Données d'essais : classer, $F_i = i/N$, tracer $\ln\ln(1/S)$ contre $\ln\sigma$ :
        pente $m$.
\end{enumerate}
\end{exbox}

\begin{exbox}[title={Lire un diagramme de phase}]
\begin{enumerate}
  \item Placer le point (composition, température) ; s'il est dans un domaine biphasé, tracer
        l'horizontale jusqu'aux frontières : compositions des deux phases.
  \item Levier : la fraction d'une phase est le bras \textbf{opposé} / longueur totale.
  \item Pour les microstructures (\og combien d'eutectique ?\fg{}, \og combien de perlite ?\fg{}),
        appliquer le levier \textbf{juste au-dessus} du palier eutectique ou eutectoïde.
  \item Se demander si la diffusion a eu le temps : non pour les solutés de substitution
        (microségrégation), oui pour C dans Fe.
\end{enumerate}
\end{exbox}

\begin{exbox}[title={Choisir un traitement thermique}]
\begin{itemize}
  \item Alliage d'Al durcissable (2xxx, 6xxx, 7xxx) : mise en solution (sous l'eutectique !),
        trempe, revenu au temps optimal.
  \item Métal écroui trop dur : recuit (restauration, recristallisation).
  \item Acier : austénitiser, tremper en évitant le nez du TTT (sinon perlite/bainite), revenir ;
        pièce épaisse : choisir un acier à forte trempabilité ; surface dure : cémentation,
        nitruration, trempe superficielle.
\end{itemize}
\end{exbox}

\section{Les pièges classiques}

\begin{warnbox}[title={À ne pas oublier le jour de l'examen}]
\begin{itemize}
  \item $\sigma = R(1+e)$ suppose le \textbf{volume constant} (plasticité), et n'a plus de sens
        après le début de la striction.
  \item $R_m$ n'est pas un affaiblissement du métal : c'est le début de la striction.
  \item Fissure interne : $a$ = \textbf{demi}-longueur.
  \item $\sigma_0$ de Weibull n'est pas la médiane ($S = 1/e$) ; $V_0$ et $\sigma_0$ vont par paire.
  \item La flexion \og flatte\fg{} les céramiques ($k_m < 1$).
  \item Le $1-\nu$ des contraintes thermiques vient de l'état biaxial ; la traction est en
        \textbf{surface} au refroidissement brutal.
  \item Le levier : bras opposé. Le diagramme de phase ne décrit pas une pièce brute de coulée
        (microségrégation).
  \item Homogénéiser \textbf{sous} la température eutectique (sinon brûlure).
  \item Le durcissement structural a un \textbf{optimum} ; au-delà, sur-revenu.
  \item On trempe pour \textbf{éviter} perlite et bainite. La trempabilité n'est pas la dureté.
  \item La martensite est athermale : attendre à $T$ fixée ne la fait pas progresser.
  \item La transition ductile--fragile concerne les métaux \textbf{CC}, pas les CFC.
  \item Hall--Petch est le seul mécanisme qui ne coûte pas de ductilité.
\end{itemize}
\end{warnbox}
