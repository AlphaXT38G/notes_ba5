\section{From Numbers to Vectors and Matrices}


Recap of lecture 1: we understand how floating point arithmetic generally works (play with
formats on \href{https://evanw.github.io/float-toy/}{Float Toy}). \textbf{But} individual numbers
aren't enough: real problems involve millions to billions of numbers arranged in specific ways.
Today: the most common arrangement, \emph{vectors \& matrices}.


\begin{definition}

\textbf{SCALARS, VECTORS, MATRICES, TENSORS}

Key numerical data structures organized by dimension: scalar, row vector, column vector, matrix,
tensor. All are \textbf{nd-arrays} (general n-dimensional storage); vectors and matrices are
representations of \emph{linear functions}.

\end{definition}


\begin{definition}

\textbf{LINEAR FUNCTION}

$f$ is a linear function \emph{if and only if}

\begin{align*}
f(\mathbf{x} + \mathbf{y}) &= f(\mathbf{x}) + f(\mathbf{y}) && \text{(additivity)} \\
f(\alpha\mathbf{x}) &= \alpha f(\mathbf{x}) \quad (\alpha \in \mathbb{R}) && \text{(homogeneity)}
\end{align*}

In this case $f(\mathbf{x}) = \mathbf{A}\mathbf{x}$ fully encodes everything that $f$ does!

\end{definition}


\begin{formula}

\textbf{Matrix-vector multiplication}

\[
(\mathbf{A}\mathbf{x})_i = \sum_{k=1}^{n} A_{ik} x_k
\]

With the columns $\mathbf{a}^{(1)}, \mathbf{a}^{(2)}, \mathbf{a}^{(3)}$ of $\mathbf{A}$:

\[
\mathbf{A}\mathbf{x} = \mathbf{a}^{(1)} x_1 + \mathbf{a}^{(2)} x_2 + \mathbf{a}^{(3)} x_3
\]

a linear combination of the columns (if they are linearly dependent, they span e.g.\ only a 2D
subspace).

\end{formula}


\textbf{Basic matrix transformations in 2D}

\begin{center}

\includegraphics[
width=0.85\linewidth
]{../images/week02_01.png}

\end{center}


\begin{important}

\textbf{Why care about linear functions?}

\begin{itemize}

\item Solving linear problems is one of the (few) numerical problems that we know to solve
\textbf{really well}: when you can turn something into a linear system, good job, you are done.
\emph{Nonlinear} methods are \emph{fragile}.

\item Almost everything boils down to a linear system at some point.

\end{itemize}

\end{important}


\textbf{Course topics} (``veggies first, then the cake''): 1.\ floating point arithmetic \& error;
2.\ linear systems, LU (\emph{today}); 3.\ conditioning \& least squares basics; 4.\ least squares,
QR; 5.\ writing efficient numerical code; 6.\ eigenanalysis, SVD; 7.\ nonlinear problems; 8.\
automatic differentiation; 9.\ neural networks.

\section{Cost of Matrix Operations}


\begin{formula}

\textbf{Matrix-vector multiplication}: $\mathcal{O}(n^2)$

$\mathbf{b} \in \mathbb{R}^{10000}$, $\mathbf{A} \in \mathbb{R}^{10000 \times 10000}$ (100
million entries), measured on an M1 Mac:

\begin{center}
\small
\begin{tabular}{l|rr}
Code & Time & \\
\hline
\texttt{b = A @ x} & 11 ms & \\
rows: \texttt{b[i] = A[i, :] @ x} & 66 ms & $\sim$6$\times$ \\
cols: \texttt{b += A[:, j] * x[j]} & 309 ms & $\sim$28$\times$ \\
2 loops: \texttt{b[i] += A[i, j] * x[j]} & 30'282 ms & $\sim$2'752$\times$ \\
\end{tabular}
\end{center}

\end{formula}


\begin{definition}

\textbf{MATRIX MULTIPLICATION}

\[
(\mathbf{A}\mathbf{B})_{ij} = \sum_{k=1}^{n} A_{ik} B_{kj}
\]

Why this definition? If $f(\mathbf{x}) \leftrightarrow \mathbf{A}\mathbf{x}$ and
$g(\mathbf{x}) \leftrightarrow \mathbf{B}\mathbf{x}$, then the composition
$g(f(\mathbf{x})) \leftrightarrow \mathbf{B}\mathbf{A}\mathbf{x} = \mathbf{C}\mathbf{x}$.

\end{definition}


\begin{formula}

\textbf{Matrix-matrix multiplication}: $\mathcal{O}(n^3)$

$\mathbf{A}, \mathbf{B} \in \mathbb{R}^{1000 \times 1000}$ (1 million entries):

\begin{center}
\small
\begin{tabular}{l|rr}
Code & Time & \\
\hline
\texttt{C = A @ B} & 19 ms & \\
loop: \texttt{C[:, j] = A @ B[:, j]} & 236 ms & $\sim$6$\times$ \\
triple loop & 341'495 ms & $\sim$17'973$\times$ \\
\end{tabular}
\end{center}

\end{formula}


\begin{important}

\textbf{Is $\mathcal{O}(n^3)$ really true?} Volker Strassen, 1969: $O(n^{2.8074})$. Virginia
Williams et al., 2023: $O(n^{2.371552})$. Still an active area of research: quite possible that
the true cost is $O(n^2)$, \emph{nobody knows!} Many works are \emph{galactic} algorithms; in
practice, the naïve algorithm tends to be fastest.

\end{important}

\section{Matrix Zoo and Sparse Matrices}


\textbf{Matrix zoo}: dense, symmetric, diagonal, tridiagonal, permutation, upper triangular,
lower triangular, block, low rank, sparse. Also important: positive definite \& orthogonal
matrices.

\begin{center}

\includegraphics[
width=0.85\linewidth
]{../images/week02_02.png}

\end{center}


\begin{definition}

\textbf{SPARSE MATRIX}

\emph{Almost all} ($> 99\%$) entries are zero.

\begin{itemize}

\item Useful: can avoid storing the zeros \& skip them during computation.

\item Almost every matrix algorithm has an analogous \emph{sparse} version (or several!).

\item Encodings: \emph{Compressed Sparse Row} (CSR) or \emph{Compressed Sparse Column} (CSC)
(\texttt{scipy.sparse.csr\_matrix} \& \texttt{scipy.sparse.csc\_matrix}).

\end{itemize}

\end{definition}


\begin{example}

\begin{itemize}

\item Protein modeling, social networks, triangle meshes of 3D objects: sparsity arises because
most things \emph{aren't} connected.

\item Discretizing continuous equations: 1D heat equation
$\frac{\partial^2 f(x)}{\partial x^2} = 0$ becomes
$-f(x_{i-1}) + 2f(x_i) - f(x_{i+1}) = 0$, one row
$(0\ 0\ 0\ {-1}\ 2\ {-1}\ 0\ 0\ 0)$ of a HUGE matrix. Intuition: heat wants to spread out and
become uniform, can't have a hotspot.

\end{itemize}

\end{example}


\textbf{Sparse matrix-vector multiplication (CSR)}

\begin{center}

Row $i$ uses entries \texttt{row\_ptr[i]} to \texttt{row\_ptr[i+1]} of \texttt{col} and

\end{formula}

\section{Linear Systems}


Today: dense, upper triangular, lower triangular and permutation matrices.


\begin{definition}

\textbf{LINEAR SYSTEM}

\[
\mathbf{A}\mathbf{x} = \mathbf{b}, \qquad
\mathbf{A} \in \mathbb{R}^{m \times n}, \ \mathbf{x} \in \mathbb{R}^n, \
\mathbf{b} \in \mathbb{R}^m
\]

Can $\mathbf{b}$ be expressed as a linear combination of the columns of $\mathbf{A}$? The
solution may not exist or may not be unique! This lecture: the \emph{square} case ($m = n$).

\end{definition}


\begin{important}

\textbf{Geometric interpretation}

\begin{itemize}

\item \textbf{Column-based view}: try to reach point $\mathbf{b}$ using a linear combination of
basis vectors $\mathbf{a}^{(i)}$ (columns). $\mathbf{x}$ tells us how much of each axis we need.

\item \textbf{Row-based view}: compute the intersection of planes with normals $\mathbf{a}^{(i)}$
(rows) and offset $b_i$: $\mathbf{a}^{(i)} \cdot \mathbf{x} = b_i$.

\end{itemize}

\end{important}


\begin{warning}

\textbf{Pure versus numerical mathematics}: concepts like invertibility become more nuanced.

\[
\begin{pmatrix} 1 & 0 \\ 1 & \epsilon \end{pmatrix}
\begin{pmatrix} x_1 \\ x_2 \end{pmatrix}
= \begin{pmatrix} -1 \\ 1 \end{pmatrix}
\]

Conceptually the matrix is nonsingular for every $\epsilon \ne 0$ and singular for
$\epsilon = 0$; in IEEE754 FP it becomes gradually ``more singular'' from
$\epsilon = 10^{-1}$ to $10^{-10}$, $10^{-50}$, $0$.

\end{warning}

\section{Gaussian Elimination}


\begin{definition}

\textbf{GAUSSIAN ELIMINATION} (MATH-111 style)

On the augmented matrix $[\,\mathbf{A} \mid \mathbf{b}\,]$, three ingredients:

\begin{itemize}

\item \textbf{permute rows};

\item \textbf{scale rows};

\item \textbf{add a (scaled) version of a row} to another.

\end{itemize}

\end{definition}


\begin{example}

Add $-5\times$ row 1 to row 2:
\[
\left[\begin{array}{ccc|c} 1 & 0 & 3 & 1 \\ 5 & 1 & 1 & 2 \\ 4 & 2 & 0 & 3 \end{array}\right]
\rightarrow
\left[\begin{array}{ccc|c} 1 & 0 & 3 & 1 \\ 0 & 1 & -14 & -3 \\ 4 & 2 & 0 & 3 \end{array}\right]
\]

\end{example}


\begin{important}

Each step uses a \textbf{pivot element} on the diagonal. Observation: the values of $\mathbf{b}$
are updated but don't really influence the process.

\end{important}


\begin{warning}

Instabilities can occur even for nonzero elements (e.g.\ a tiny pivot $\varepsilon$): the pivot
element must be chosen carefully.

\textbf{Practical solution}: pick the element with the largest absolute value.

\end{warning}

\section{Solving Repeatedly: Don't Invert}


A common situation: the same system must be solved over and over again,
$\mathbf{A}\mathbf{x}^{(1)} = \mathbf{b}^{(1)}$, $\mathbf{A}\mathbf{x}^{(2)} = \mathbf{b}^{(2)}$,
\dots\ Can we \emph{precompute} something?


\begin{warning}

\textbf{Anti-pattern -- don't do this}: precompute $\mathbf{A}^{-1}$, then one matrix-vector
multiplication per solve, $\mathbf{x}^{(i)} = \mathbf{A}^{-1}\mathbf{b}^{(i)}$.

\begin{enumerate}

\item Computation of the inverse can be numerically unstable.

\item Inverse + matrix multiplies require slightly more FLOPs than the alternative (LU).

\item Generally, the inverse of a sparse matrix is dense: will run out of memory.

\end{enumerate}

\end{warning}

\section{LU Factorization}


\begin{definition}

\textbf{LU FACTORIZATION} (our first decomposition)

Run Gaussian elimination on $\mathbf{A}$ alone (no right-hand side $\mathbf{b}$), keeping track
of the transformations it performs. They can be written as matrices $\mathbf{L}$ (lower
triangular) and $\mathbf{U}$ (upper triangular):

\[
\mathbf{A} = \mathbf{L}\mathbf{U}
\]

Then ``apply'' $\mathbf{L}$ and $\mathbf{U}$ to any right-hand side $\mathbf{b}$ very
efficiently.

\end{definition}


\begin{formula}

\textbf{A single elimination step is a matrix}:

\[
\begin{bmatrix} 1 & 0 & 0 \\ c & 1 & 0 \\ 0 & 0 & 1 \end{bmatrix}
\mathbf{A}
\ \text{adds } c \times \text{row 1 to row 2}
\]

Its inverse is simple:

\[
\begin{bmatrix} 1 & 0 & 0 \\ c & 1 & 0 \\ 0 & 0 & 1 \end{bmatrix}^{-1}
= \begin{bmatrix} 1 & 0 & 0 \\ -c & 1 & 0 \\ 0 & 0 & 1 \end{bmatrix}
\]

A lower triangular matrix; the product of lower-triangular matrices is always lower-triangular.

\end{formula}


\begin{formula}

\textbf{Putting things together}: $\mathbf{U}$ is the traditional output of Gaussian
elimination, $\mathbf{L}$ reconstructs $\mathbf{A}$:

\begin{align*}
\mathbf{U} &= \mathbf{E}_6\mathbf{E}_5\mathbf{E}_4\mathbf{E}_3\mathbf{E}_2\mathbf{E}_1\mathbf{A}
\\
\mathbf{L} &= (\mathbf{E}_6\mathbf{E}_5\mathbf{E}_4\mathbf{E}_3\mathbf{E}_2\mathbf{E}_1)^{-1} \\
&= \mathbf{E}_1^{-1}\mathbf{E}_2^{-1}\mathbf{E}_3^{-1}\mathbf{E}_4^{-1}\mathbf{E}_5^{-1}
\mathbf{E}_6^{-1}
\end{align*}

\end{formula}


\begin{example}

\[
\mathbf{A} = \begin{bmatrix} 1 & 1 \\ 2 & 4 \end{bmatrix}, \quad
\mathbf{E}_1 = \begin{bmatrix} 1 & 0 \\ -2 & 1 \end{bmatrix} \ (\text{row 2} -= 2 \times
\text{row 1})
\]
\[
\mathbf{U} = \mathbf{E}_1\mathbf{A} = \begin{bmatrix} 1 & 1 \\ 0 & 2 \end{bmatrix}, \quad
\mathbf{L} = \mathbf{E}_1^{-1} = \begin{bmatrix} 1 & 0 \\ 2 & 1 \end{bmatrix}
\ \rightarrow \ \mathbf{A} = \mathbf{L}\mathbf{U}
\]

\end{example}


\begin{warning}

We won't look at low-level details of how the LU factorization is computed (ditto for others).

\end{warning}

\section{Using an LU Decomposition}


\begin{formula}

\[
\mathbf{x} = \mathbf{U}^{-1}(\mathbf{L}^{-1}\mathbf{b})
\]

Important: here, $\cdot^{-1}$ does \emph{not} mean computing a matrix inverse, but solving
triangular systems (forward substitution with $\mathbf{L}$, back substitution with
$\mathbf{U}$).

\end{formula}


\begin{important}

Forward/back-substitution have the same complexity as matrix-vector multiplication: solving a
triangular system takes $O(n^2)$, i.e.\ \emph{all the hard work is done already!}

\end{important}


\begin{definition}

\textbf{LU WITH PIVOTING} (column pivoting, also known as partial pivoting)

\[
\mathbf{A} = \mathbf{P}\mathbf{L}\mathbf{U}, \qquad
\mathbf{x} = \mathbf{U}^{-1}(\mathbf{L}^{-1}(\mathbf{P}^{-1}\mathbf{b}))
\]

$\mathbf{P}$ is a \textbf{permutation matrix}: each row/column contains a single 1, the rest is
zero.

\end{definition}


\begin{example}

\textbf{In SciPy}

\begin{verbatim}
import scipy.linalg as la
lu, piv = la.lu_factor(A)
x = la.lu_solve((lu, piv), b)
\end{verbatim}

Returns a pivot element vector and a single 2D array: the diagonal elements of $\mathbf{L}$ are
1, so both components can be packed tightly.

\end{example}
