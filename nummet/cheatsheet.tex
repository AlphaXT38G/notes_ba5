\begin{thm}[title=Condition number,halign=center,top=3pt,bottom=3pt]
\(\displaystyle
\colorbox{greenColor!20}{\(\dfrac{\norm{\Delta\vect{x}}}{\norm{\vect{x}}}\)}
\le \underbrace{\lVert A^{-1}\rVert\,\norm{A}}_{=\,\kappa(A)}
\colorbox{blueColor!20}{\(\dfrac{\norm{\Delta\vect{b}}}{\norm{\vect{b}}}\)}\)\\[2pt]
\(\operatorname{cond}(A) = \kappa(A) = \norm{A}\,\lVert A^{-1}\rVert\)
\end{thm}

\begin{defn}[title=Intuition]
\(A\): circle \(\to\) ellipse\\
\(\kappa=\) max stretch / min stretch \(=\sigma_{\max}/\sigma_{\min}\)\\
\(\kappa=1\): still a circle. Flat ellipse: \(\kappa\gg1\)\\
Worst: \(\vect{x}\) on max, \(\Delta\vect{b}\) on min stretch \(\Rightarrow\) error \(\times\,\kappa\)\\
\(\det\) = volume, \(\kappa\) = shape\\
Lose \(\approx\log_{10}\kappa\) digits (of \(\approx16\) in double)
\end{defn}

\begin{prop}[title=Properties]
\(\operatorname{cond}(A)=\infty \iff A\) singular\\
\(\operatorname{cond}(I)=1\)\\
\(\operatorname{cond}(\gamma A)=\operatorname{cond}(A),\ \gamma\neq0\)\\
\(\operatorname{cond}(QA)=\operatorname{cond}(A),\ Q\) rotation
\end{prop}

\begin{warn}[title=Not the determinant]
\(A=\tfrac{1}{10}I\in\R^{1000\times1000}\): \(\det(A)=10^{-1000}\), but
\(\operatorname{cond}(A)=1\) (trivial to solve).
\end{warn}

\begin{thm}[title=Implications (good solver{,} exact \(\vect{b}\)),halign=center,top=3pt,bottom=3pt]
\(\displaystyle \frac{\norm{\Delta\vect{x}}}{\norm{\vect{x}}} \lesssim
\operatorname{cond}(A)\,\operatorname{ulp}(1)\)\\[2pt]
\(\operatorname{ulp}(1)\approx2^{-23}\) (single), \(\approx2^{-52}\) (double)
\end{thm}
