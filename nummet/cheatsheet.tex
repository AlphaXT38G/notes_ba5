\section{LU Factorization}
\begin{defn}[title={LU / PLU}]
\begin{center}
\includegraphics[width=0.6\linewidth]{images/lu.png}
\end{center}
\(A=LU\): \(L\) lower-triangular (unit diag.), \(U\) upper-triangular = Gaussian elimination.
\(U=E_k\cdots E_1A\), \(L=E_1^{-1}\cdots E_k^{-1}\).\\
\textbf{Pivoting:} \(A=PLU\), \(P\) permutation.
\end{defn}

\begin{prop}[title={Solve $Ax=b$}]
\(x=U^{-1}(L^{-1}(P^{-1}b))\): forward then back substitution (no inverse computed).\\
Factor once, then \(O(n^2)\) per right-hand side.
\end{prop}

\begin{defn}[title=Cholesky (SPD)]
\(A=LL^T\): half the cost/storage of LU.
\end{defn}
