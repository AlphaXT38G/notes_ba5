\section{Course Essentials}


\begin{summary}

\begin{itemize}

\item CS-290, C\'ecile Hardebolle. \textbf{Flipped class}: theory at home (videos + quizzes,
about 5 videos/week), practice in class.

\item \textbf{Tuesday 10:15-12:00}: programming exercise (Jupyter notebook on Noto).

\item \textbf{Monday 15:15-17:00}: review questions (ttpoll) + case studies.

\item Solutions are provided for notebooks and case studies.

\end{itemize}

\end{summary}


\textbf{Contents of the course}

\begin{itemize}

\item \textbf{Safety}: harmful impacts, safety \& security, micro/meso/macro.

\item \textbf{Fairness}: sources and impacts of biases, fairness measures.

\item \textbf{Sustainability}: environmental impact, life cycle assessment.

\item \textbf{Empowerment}: user agency, transparency, explainability.

\item \textbf{Privacy}: data traces, data protection, re-identification, surveillance.

\item \textbf{Strategies}: stakeholder analysis, impact anticipation, values elicitation, risk
assessment, decision-making.

\end{itemize}


\begin{goat}

You watch the videos at home and you practise in class, with help. The course is about the
ways software hurts people, and about methods to see it coming.

\end{goat}

\section{Evaluation}


\begin{important}

\begin{itemize}

\item \textbf{Mini-project 5\%}: groups of 2, 4 deliverables (2-3 pages), no programming,
one in four graded. \textbf{Questions on it at the final exam.}

\item \textbf{Mid-term 25\%}: 2 November, 15:15, 1h30, pen and paper, individual,
\textbf{one A4 sheet} (both sides, free format). Chapters Introduction, Safety 1-2, Fairness
1-2. Single-choice + short open questions; notebooks (code reading, output, reflection) and
theory.

\item \textbf{Final 70\%}: winter session, same rules, whole course.

\end{itemize}

\end{important}


\begin{warning}

\textbf{GenAI is prohibited for all graded work.} Answers can be in French. Arguments must be
understandable: 1 point $\approx$ 1-2 complete sentences.

\end{warning}


\textbf{Learning loop}: \textbf{Experience} (problem to solve with programming) +
\textbf{Knowledge} (theory + strategies) + \textbf{Habits} (strategies applied to cases).


\begin{goat}

Three grades: a tiny project, a mid-term and a big final, always with one cheat sheet and no
ChatGPT.

\end{goat}

\section{AI Incidents and Ethical Debt}


\begin{example}

\textbf{OECD AI Incidents and Hazards Monitor (AIM)}

\begin{itemize}

\item Collects AI incidents (harm happened) and hazards (harm could happen) reported in the
press.

\item The \textbf{number} of incidents and hazards is \textbf{increasing}; their
\textbf{proportion} among AI events is rather stable.

\item Limits: depends on what the press reports (countries, languages, spectacular cases).

\end{itemize}

\end{example}


\begin{definition}

\textbf{ETHICAL DEBT} (Petrozzino, 2021)

The \textbf{price paid by people and the environment} because of systems
\textbf{released with ethical flaws}.

\end{definition}
