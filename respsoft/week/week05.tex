\section{From Micro to Macro}


\begin{definition}

\textbf{LEVELS OF ANALYSIS}

\begin{itemize}

\item \textbf{Micro}: individuals.

\item \textbf{Meso}: groups, communities, organisations.

\item \textbf{Macro}: societies, economies, planet.

\end{itemize}

Macro = used \textbf{by many}, \textbf{in many places}, \textbf{all the time}, \textbf{for a
long time}.

\end{definition}


\begin{important}

Internet = double-edged sword (Brey 2019): access to information, communities, commerce
$\leftrightarrow$ addiction, overload, harm to democracy. Macro impacts are
\textbf{controversial}: hard to measure, causality hard to prove.

\textbf{Problem of one person}: involve stakeholders at all levels, work with social sciences,
choose your employer.

\end{important}


\begin{example}

\textbf{GenAI and learning} (macro): 26\,811 students, 30 months: homework $+18\%$, time
$-30\%$, but exams $-20\%$; worst for junior students and those who outsource.

\end{example}


\begin{goat}

One goat eating a flower is fine; the whole herd every day for years kills the meadow.

\end{goat}

\section{Dis- and Mis-information}


\begin{definition}

\begin{itemize}

\item \textbf{Misinformation}: false, \textbf{no} intent.

\item \textbf{Disinformation}: false, \textbf{with} intent.

\item \textbf{Malinformation}: \textbf{true}, used to harm.

\item \textbf{Fake news}: false, in the \textbf{style of news}.

\end{itemize}

\end{definition}


\begin{formula}

\textbf{Why we believe} (Ecker 2022)

\textbf{Cognitive}: System 1 (fast) vs. 2 (slow); intuitive thinking; \textbf{illusory truth}
(repetition $\Rightarrow$ belief).

\textbf{Socio-affective}: \textbf{source cues} (credibility, power, in-group),
\textbf{false consensus}, emotions, worldview.

\textbf{Correction}: \textbf{continued influence effect}.

\end{formula}


\begin{example}

\begin{itemize}

\item A friend: carrots improve night vision $\rightarrow$ \textbf{misinformation}.

\item Still believing after the debunk $\rightarrow$ \textbf{continued influence}.

\item Believing Elon Musk's post $\rightarrow$ \textbf{source cues}.

\item ``Fight'' boosts reposts 20\% $\rightarrow$ \textbf{intuitive reactions}.

\end{itemize}

\end{example}


\textbf{WEF 2024}: mis/disinformation = most severe global risk for the next 2 years.
