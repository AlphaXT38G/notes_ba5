\section{Axiomes de probabilité}


\begin{definition}

\begin{itemize}

\item $\zeta_i$ : résultat de la $i$-ème expérience.

\item Évènement $A, B, \dots$ : un ensemble de résultats $\{\text{quelques } \zeta_i\}$.

\item $\Omega = \{\text{tous les } \zeta_i\}$ : évènement certain ; $\emptyset$ : évènement
impossible.

\end{itemize}

La probabilité $P(A)$ est un réel qui satisfait :

\begin{itemize}

\item \textbf{A1.} $P(A) \ge 0$

\item \textbf{A2.} $P(\Omega) = 1$

\item \textbf{A3.} ($\Omega$ fini) si $A \cap B = \emptyset$, alors $P(A\cup B) = P(A) + P(B)$

\item \textbf{A3'.} ($\Omega$ infini) si $A_i \cap A_j = \emptyset$ pour $i \ne j$, alors
$P\big(\bigcup_{i=1}^\infty A_i\big) = \sum_{i=1}^\infty P(A_i)$

\end{itemize}

\end{definition}


\begin{formula}

Corollaires (avec $\bar A = \Omega \setminus A$) :

\[
P(\bar A) = 1 - P(A) \qquad P(A) \le 1 \qquad P(\emptyset) = 0
\]

\[
P(A \cup B) = P(A) + P(B) - P(A \cap B)
\]

\[
A \subseteq B \;\Rightarrow\; P(A) \le P(B)
\]

\end{formula}


\begin{example}

\textbf{Preuve de $P(\bar A) = 1 - P(A)$} (slides) : $\Omega = A \cup \bar A$ donc
$P(A\cup\bar A) = 1$, et $A \cap \bar A = \emptyset$ donc $P(A \cup \bar A) = P(A) +
P(\bar A)$ par A3.

\vspace{3pt}

\textbf{Preuve de $P(A\cup B)$} : on découpe en morceaux disjoints, $A \cup B = (A\cap\bar
B) \cup (A \cap B) \cup (\bar A \cap B)$, avec $P(A) = P(A\cap\bar B) + P(A\cap B)$ et $P(B) =
P(\bar A\cap B) + P(A\cap B)$. En additionnant, $A\cap B$ est compté deux fois : on le retire
une fois.

\end{example}


\begin{simple}

Une probabilité, c'est une \textbf{quantité de « chance » à partager}, il y en a exactement 1
en tout. Elle n'est jamais négative, et si deux évènements ne peuvent pas arriver ensemble,
leurs chances s'additionnent. Pour $A \cup B$ : si on additionne les deux, la partie commune
$A\cap B$ est comptée deux fois, donc on l'enlève une fois.

\end{simple}

\section{Probabilité conditionnelle et indépendance}


\begin{definition}

\textbf{Probabilité conditionnelle} de $A$ sachant que $B$ s'est produit :

\[
P(A|B) = \frac{P(A\cap B)}{P(B)} \quad \text{si } P(B) \ne 0,
\]

et $P(A|B) = 0$ si $P(B) = 0$.

\textbf{Partition} de $\Omega$ : $[A_1,\dots,A_n]$ tels que $A_1\cup\dots\cup A_n = \Omega$ et
$A_i\cap A_j = \emptyset$ pour $i\ne j$.

\end{definition}


\begin{formula}

\textbf{Théorème des probabilités totales :}

\[
P(B) = \sum_{i=1}^n P(B|A_i)\,P(A_i)
\]

\textbf{Règle de Bayes :}

\[
P(A_i|B) = \frac{P(B|A_i)\,P(A_i)}{\sum_{j=1}^n P(B|A_j)\,P(A_j)}
\]

\end{formula}


\begin{example}

\textbf{Preuve des probabilités totales} (exercice 1) : $B = B\cap\Omega = B \cap (A_1 \cup
\dots \cup A_n) = (B\cap A_1) \cup \dots \cup (B \cap A_n)$, des morceaux disjoints, donc
$P(B) = \sum_i P(B\cap A_i) = \sum_i P(B|A_i)P(A_i)$.

\vspace{3pt}

\textbf{Preuve de Bayes} (exercice 2) : $P(A_i\cap B) = P(A_i|B)P(B) = P(B|A_i)P(A_i)$, puis on
remplace $P(B)$ par les probabilités totales.

\end{example}


\begin{definition}

\textbf{Indépendance.} $A$ et $B$ sont indépendants si $P(A\cap B) = P(A)\,P(B)$ (alors
$P(A|B) = P(A)$ et $P(B|A) = P(B)$).

\vspace{3pt}

$A$, $B$, $C$ sont indépendants si et seulement si \textbf{les quatre} conditions sont
vraies :

\[
P(A \cap B) = P(A)P(B), \quad P(A\cap C) = P(A)P(C),
\]
\[
P(B\cap C) = P(B)P(C),
\]
\[
P(A\cap B\cap C) = P(A)P(B)P(C).
\]

\end{definition}


\begin{warning}

Indépendants \textbf{deux à deux} ne suffit pas. Exemple des slides : trois évènements de
probabilité $\frac12$ avec $P(A\cap B) = P(A\cap C) = P(B\cap C) = \frac14$ (chaque paire est
indépendante), mais $A\cap B\cap C = \emptyset$, donc $P(A\cap B\cap C) = 0 \ne \frac18$.

\end{warning}


\begin{simple}

$P(A|B)$, c'est : « on \textbf{sait} que $B$ est arrivé, on ne regarde plus que le monde où $B$
est vrai ». On se restreint à $B$ et on renormalise (on divise par $P(B)$).

\vspace{3pt}

\textbf{Probabilités totales} : pour savoir la chance de $B$, on regarde chaque cas possible
$A_i$ séparément, et on fait la moyenne pondérée. \textbf{Bayes} : on retourne la flèche, de
« la cause donne l'effet » à « je vois l'effet, quelle était la cause ? ».

\vspace{3pt}

\textbf{Indépendants} : savoir que l'un est arrivé ne change rien à la chance de l'autre.

\end{simple}

\section{Variable aléatoire et fonction de répartition}


\begin{definition}

Une \textbf{variable aléatoire} (v.a.) est une fonction qui associe à chaque résultat $\zeta$
un réel $X(\zeta)$, telle que :

\begin{itemize}

\item \textbf{C1.} $\{\zeta \mid X(\zeta) \le x\}$ est un évènement pour tout $x\in\mathbb R$ ;

\item \textbf{C2.} $P(X = \infty) = P(X = -\infty) = 0$.

\end{itemize}

$S_X = \{X(\zeta) \mid \zeta\in\Omega\}$ : l'ensemble des valeurs que $X$ peut prendre.

\end{definition}


\begin{formula}

\textbf{Fonction de répartition} (CDF) :

\[
F_X(x) = P(X \le x)
\]

\begin{itemize}

\item $0 \le F_X(x) \le 1$, $\lim_{x\to-\infty}F_X(x) = 0$, $\lim_{x\to\infty}F_X(x) = 1$ ;

\item croissante : $a < b \Rightarrow F_X(a) \le F_X(b)$ ;

\item $P(a < X \le b) = F_X(b) - F_X(a)$ ;

\item continue à droite : $F_X(x) = F_X(x^+)$.

\end{itemize}

\end{formula}


\begin{definition}

\begin{itemize}

\item \textbf{V.a. discrète} : $S_X = \{x_1, x_2, \dots\}$ dénombrable. $F_X$ est en
\textbf{escalier}, avec un saut en chaque $x_i$ : $p_i = P(X = x_i) = F_X(x_i) - F_X(x_i^-)$.

\item \textbf{V.a. continue} : $F_X$ continue. Alors $P(X = x) = 0$ pour tout $x$.

\item \textbf{V.a. mixte} : $F_X$ discontinue mais pas en escalier (un peu des deux).

\end{itemize}

\end{definition}


\begin{simple}

Une v.a., c'est juste \textbf{un nombre qui sort d'une expérience} : on lance un dé, la v.a.
dit « 4 ». $F_X(x)$ répond à une seule question : « quelle est la chance que le nombre soit
\textbf{au plus} $x$ ? ». Elle monte de 0 à 1. Si $X$ ne prend que quelques valeurs, elle
monte par marches (discrète) ; si $X$ peut prendre n'importe quelle valeur, elle monte en
douceur (continue), et tomber pile sur une valeur précise a une chance nulle.

\end{simple}

\section{Densité de probabilité}


\begin{formula}

\[
f_X(x) = \frac{dF_X(x)}{dx}
\]

\begin{itemize}

\item $f_X(x) \ge 0$ et $\int_{-\infty}^{\infty} f_X(x)\,dx = 1$ ;

\item $F_X(a) = P(X\le a) = \int_{-\infty}^a f_X(x)\,dx$ ;

\item $P(a < X \le b) = \int_a^b f_X(x)\,dx$, donc $P(x < X \le x+\Delta x) \approx
f_X(x)\Delta x$ quand $\Delta x \to 0$.

\end{itemize}

\end{formula}


\begin{formula}

\textbf{Cas discret}, avec les impulsions de Dirac et $p_i = P(X = x_i)$ :

\[
f_X(x) = \sum_i p_i\,\delta(x - x_i)
\]

Alors $p_i \ge 0$, $\sum_i p_i = 1$, $F_X(a) = \sum_{x_i \le a} p_i$.

\end{formula}


\begin{simple}

La densité $f_X(x)$ n'est \textbf{pas} une probabilité : c'est une probabilité \textbf{par
unité de longueur}. La probabilité, c'est l'\textbf{aire} sous la courbe. Comme un tas de
sable étalé sur une ligne : là où le tas est haut, $X$ tombe souvent ; la quantité de sable
entre $a$ et $b$ est la chance que $X$ tombe entre $a$ et $b$, et il y a 1 kilo de sable en
tout. Pour une v.a. discrète, le sable est en petits tas pointus (les Dirac) posés sur les
valeurs $x_i$.

\end{simple}
