\section{PAM bruité : la chaîne de transmission}


\begin{definition}

Transmission binaire en \textbf{bande de base} (sans porteuse) : les bits $A_k$ (indépendants,
$+1$ avec probabilité $p$ pour le bit « 1 », $-1$ avec probabilité $1-p$ pour le bit « 0 »)
sont mis en forme par le pulse rectangulaire $g(t) = a\,\mathbb 1_{[0,T]}(t)$, passent dans un
canal, auquel s'ajoute un \textbf{bruit blanc gaussien} $N(t)$ ($S_N = N_0/2$) indépendant de
$X$, puis dans un filtre de réception, sont échantillonnés toutes les $T$ et un
\textbf{détecteur} à seuil $\gamma$ décide « 1 » si l'échantillon est $\ge\gamma$, « 0 » sinon.

\vspace{3pt}

Cas le plus simple : canal \textbf{non dispersif} ($H(f) = 1$), signaux synchronisés :

\[
Y(t) = \sum_kA_k\,a\,\mathbb 1_{[0,T]}(t-kT) + N(t)
\]

\end{definition}


\begin{formula}

Filtre adapté au pulse rectangulaire : on \textbf{intègre} $Y$ sur une période et on divise par
$T$ :

\[
Z((n+1)T) = \frac1T\int_{nT}^{(n+1)T}Y(t)\,dt
\]
\[
= aA_n + \frac1T\int_{nT}^{(n+1)T}N(t)\,dt
\]

Sachant $A_n$, $Z$ est \textbf{gaussienne}, de moyenne $\pm a$ et de variance

\[
\frac1{T^2}\iint R_N(t-s)\,dt\,ds = \frac{N_0}{2T^2}\iint\delta(t-s)\,dt\,ds = \frac{N_0}{2T}
\]

\end{formula}


\begin{simple}

Chaque bit est un créneau, à la hauteur $+a$ ou $-a$, et du bruit vient brouiller le tout. Le
récepteur fait la \textbf{moyenne} du signal pendant la durée du créneau : le créneau reste, et
le bruit, qui part dans tous les sens, se moyenne et diminue. Il reste une valeur autour de
$+a$ ou de $-a$, en forme de cloche, qu'il faut classer.

\end{simple}


%=========================================================
\section{Seuil optimal et rapport de vraisemblance}


\begin{formula}

Le récepteur optimal choisit « 1 » ($A_n = 1$, hypothèse $H_0$) si

\[
P(A_n = 1 | Z = z) > P(A_n = -1|Z = z)
\]

et « 0 » sinon. Par Bayes, cela revient à comparer le \textbf{rapport de vraisemblance}

\[
L(z) = \frac{f_{Z|A=1}(z|A_n = 1)}{f_{Z|A=-1}(z|A_n=-1)} = \exp\Big(\frac{4az}{N_0/T}\Big)
\]

à $(1-p)/p$. Le seuil optimal vérifie $L(\gamma) = (1-p)/p$ :

\[
\gamma = \frac{N_0}{4aT}\ln\frac{1-p}p
\]

\end{formula}


\begin{important}

On suppose ici que la probabilité \textbf{a priori} $p$ est connue. Si $p = 1/2$ (bits
équiprobables), $\gamma = 0$.

\end{important}


\begin{simple}

On reçoit $z$. Deux suspects : le bit 1 (cloche centrée en $+a$) ou le bit 0 (cloche centrée en
$-a$). On choisit celui qui rend $z$ \textbf{le plus probable}, en tenant compte de qui est le
plus fréquent. Si les deux bits sont aussi fréquents, la frontière est pile au milieu, en 0.
Si les 1 sont plus fréquents, on pousse la frontière vers les 0 pour leur laisser plus de
place.

\end{simple}


%=========================================================
\section{Probabilité d'erreur}


\begin{formula}

\[
P_e = (1-p)\int_\gamma^\infty f_{Z|A=-1}(z)\,dz + p\int_{-\infty}^\gamma f_{Z|A=1}(z)\,dz
\]

Avec $p = 1/2$ ($\gamma = 0$), et l'\textbf{énergie par bit} $E_b = \int g^2(t)\,dt = a^2T$ :

\[
\boxed{P_e = \frac12\operatorname{erfc}\Big(\sqrt{\frac{E_b}{N_0}}\Big) =
Q\Big(\sqrt{\frac{2E_b}{N_0}}\Big)}
\]

avec $\operatorname{erfc}(x) = \frac2{\sqrt\pi}\int_x^\infty e^{-\xi^2}d\xi = 2Q(\sqrt2x)$.

\end{formula}


\begin{important}

$P_e$ ne dépend que du rapport $E_b/N_0$ (souvent en dB) et décroît très vite quand il
augmente : de l'ordre de $10^{-2}$ vers 5 dB et de $10^{-16}$ vers 15 dB sur la figure du cours.

\vspace{3pt}

\textbf{Exercice 18} : que devient $P_e$ si $\gamma = 0$ mais $p\ne1/2$ ? quel $\gamma$
choisir sans connaître $p$ ? doubler $T$ ou doubler $a$ pour passer de $P_e = 10^{-3}$ à
$10^{-6}$ ?

\end{important}


\begin{simple}

Une erreur, c'est quand le bruit pousse la valeur reçue \textbf{du mauvais côté} de la
frontière. Plus l'énergie d'un bit est grande par rapport au bruit, plus les deux cloches sont
éloignées, et plus c'est rare. $Q(x)$, c'est juste « la chance qu'une cloche normale dépasse
$x$ ».

\end{simple}

\section{Processus à temps discret}


\begin{definition}

Le temps $t\in\mathbb R$ est remplacé par un entier $n\in\mathbb Z$ (le numéro de
l'échantillon : du temps discrétisé, mais aussi par exemple l'indice d'un pixel). Un processus
à temps discret est \textbf{une suite de v.a.} $X(n)$. Mêmes définitions :

\[
\mu_X(n) = E[X(n)] \qquad R_X(n_1,n_2) = E[X(n_1)X(n_2)]
\]
\[
C_X(n_1,n_2) = R_X(n_1,n_2) - \mu_X(n_1)\mu_X(n_2)
\]

et $R_{XY}$, $C_{XY}$, orthogonaux, non corrélés, comme au module 3.

\end{definition}


\begin{formula}

Avec $m$ échantillons $X(n_1),\dots,X(n_m)$ : \textbf{matrice de corrélation} $(\Phi_X)_{ik} =
R_X(n_i,n_k)$ et \textbf{matrice de covariance} $(\Gamma_X)_{ik} = C_X(n_i,n_k)$.

\vspace{3pt}

\textbf{Stationnarité} : mêmes définitions (SSS, WSS : $\mu_X$ constante, $R_X(n_1-n_2)$). Si
$X$ est WSS, la matrice de corrélation d'ordre $m$ est

\[
\Phi_X = \begin{pmatrix} R_X(0) & R_X(1) & \cdots & R_X(m-1) \\ R_X(1) & R_X(0) & \cdots &
R_X(m-2) \\ \vdots & & \ddots & \vdots \\ R_X(m-1) & R_X(m-2) & \cdots & R_X(0)\end{pmatrix}
\]

\textbf{symétrique}, \textbf{semi-définie positive} et \textbf{de Toeplitz} (constante le long
des diagonales, car $R_X(n,n-k) = R_X(k)$).

\end{formula}


\begin{simple}

Rien de neuf, on remplace les intégrales sur le temps par des sommes sur les cases. La matrice
de corrélation d'un processus stationnaire a une forme très régulière : chaque diagonale
contient le même nombre, parce que seul l'écart entre deux cases compte.

\end{simple}

\section{Exemples : bruit blanc, MA, AR, ARMA}


\begin{definition}

\textbf{Sinusoïde à phase uniforme} $X(n) = a\sin(2\pi f_0n + \Phi)$ : $\mu_X = 0$,
$R_X(n_1,n_2) = \frac{a^2}2\cos(2\pi f_0(n_1-n_2))$.

\vspace{3pt}

\textbf{Bruit blanc} : moyenne nulle et

\[
R_N(n_1,n_2) = \begin{cases}\sigma_N^2 & n_1 = n_2 \\ 0 & n_1 \ne n_2\end{cases}
\]

Une suite de v.a. centrées \textbf{non corrélées}. Contrairement au continu, sa variance
$\sigma_N^2$ est \textbf{finie} (le Dirac discret vaut 1 en 0).

\end{definition}


\begin{formula}

$U(n)$ : \textbf{processus d'innovation}, par défaut un bruit blanc de variance $\sigma_U^2$.

\begin{itemize}

\item \textbf{MA($m$)} (moyenne mobile) : $X(n) = \sum_{k=0}^mb_kU(n-k)$.

\item \textbf{AR($p$)} (auto-régressif), calculé récursivement à partir de $p$ valeurs
initiales, coefficients choisis pour que le système soit stable :

\[
X(n) = -\sum_{k=1}^pa_kX(n-k) + U(n)
\]

\item \textbf{ARMA($m,p$)} : $X(n) = -\sum_{k=1}^pa_kX(n-k) + \sum_{k=0}^mb_kU(n-k)$.

\item \textbf{ARIMA} : $Y(n) = Y(n-1) + X(n)$, avec $X$ ARMA (l'analogue d'une intégration).

\end{itemize}

Très utilisés pour les séries temporelles.

\end{formula}


\begin{simple}

\begin{itemize}

\item \textbf{MA} : la valeur d'aujourd'hui est un mélange des \textbf{chocs} au hasard des
derniers jours.

\item \textbf{AR} : la valeur d'aujourd'hui est un mélange des \textbf{valeurs} des jours
précédents, plus un nouveau choc. Comme la température : demain ressemble à aujourd'hui, avec
un peu de surprise.

\end{itemize}

\end{simple}

\section{Ergodisme et densité spectrale en discret}


\begin{formula}

\textbf{Moyenne temporelle} : $\langle X(n)\rangle_m = \frac1{2m+1}\sum_{n=-m}^mX(n)$, non
biaisée. \textbf{Ergodique par rapport à la moyenne} ssi

\[
\lim_{m\to\infty}\frac1{2m+1}\sum_{k=-2m}^{2m}C_X(k)\Big(1 - \frac{|k|}{2m+1}\Big) = 0
\]

Suffisant : $\lim_{k\to\infty}C_X(k) = 0$.

\end{formula}


\begin{formula}

\textbf{Densité spectrale} : transformée de Fourier \textbf{discrète} de $R_X$, périodique de
période 1, on regarde $-\frac12\le f\le\frac12$.

\[
S_X(f) = \sum_{k=-\infty}^{\infty}R_X(k)e^{-j2\pi fk}
\]
\[
R_X(k) = \int_{-1/2}^{1/2}S_X(f)e^{j2\pi fk}df
\]

En $z$ : $\hat S_X(z) = \sum_kR_X(k)z^{-k}$, et $S_X(f) = \hat S_X(e^{j2\pi f})$.

\vspace{3pt}

$E[X^2(n)] = R_X(0) = \int_{-1/2}^{1/2}S_X(f)\,df$ ; $S_X$ réelle, paire, $\ge0$.

\vspace{3pt}

\textbf{Exemples} : sinusoïde, $S_X(f) = \frac{a^2}4(\delta(f-f_0) + \delta(f+f_0))$ ; bruit
blanc, $S_N(f) = \sigma_N^2$ sur toute la bande.

\end{formula}

\section{Système LTI à temps discret}


\begin{formula}

$H(f) = \sum_lh(l)e^{-j2\pi fl}$ (si $\sum|h| < \infty$), $\hat H(z) = \sum_lh(l)z^{-l}$, $H(f) =
\hat H(e^{2\pi jf})$. Sortie $Y(n) = \sum_lh(l)X(n-l)$, WSS si $X$ l'est :

\[
E[Y(n)] = H(0)\mu_X
\]
\[
R_Y(k) = \sum_l\sum_mh(l)h(m)R_X(k+l-m)
\]
\[
S_Y(f) = |H(f)|^2S_X(f) \qquad \hat S_Y(z) = \hat H(z)\hat H(1/z)\hat S_X(z)
\]
\[
S_{YX}(f) = H(f)S_X(f) \qquad S_{XY}(f) = H^\star(f)S_X(f)
\]

\end{formula}


\begin{formula}

\textbf{ARMA} = innovation $U$ filtrée par

\[
\hat H_{\mathrm{ARMA}}(z) = \frac{\hat B(z)}{\hat A(z)} = \frac{\sum_{k=0}^mb_kz^{-k}}{1 +
\sum_{k=1}^pa_kz^{-k}}
\]
\[
S_X(f) = \sigma_U^2\frac{|\sum_{k=0}^mb_ke^{-2\pi jfk}|^2}{|1 + \sum_{k=1}^pa_ke^{-2\pi
jfk}|^2}
\]

MA : $\hat A = 1$ (que des zéros). AR : $\hat B = 1$ (que des pôles).

\vspace{3pt}

\textbf{Factorisation spectrale} : si $\int_{-1/2}^{1/2}|\ln S_X(f)|df < \infty$, il existe
$\hat G$ tel que $\hat S_X(z) = \hat G(z)\hat G(z^{-1})$, soit $S_X(f) = G(f)G^\star(f)$.

\end{formula}


\begin{simple}

Un processus ARMA, c'est du \textbf{bruit blanc passé dans un filtre}. Le bruit blanc a toutes
les fréquences à plat, et le filtre sculpte ce spectre plat : son spectre est juste $\sigma_U^2$
fois $|H|^2$. Et presque tout processus peut se voir comme ça.

\end{simple}
