\section{Paire de v.a. : répartition et densité jointes}


\begin{formula}

\textbf{Fonction de répartition jointe} :

\[
F_{XY}(x,y) = P(X\le x, Y \le y)
\]

\textbf{Marginales} : $F_X(x) = \lim_{y\to\infty} F_{XY}(x,y)$, $F_Y(y) = \lim_{x\to\infty}
F_{XY}(x,y)$.

\vspace{3pt}

\textbf{Densité jointe} :

\[
f_{XY}(x,y) = \frac{\partial^2 F_{XY}(x,y)}{\partial x\,\partial y}, \qquad \iint f_{XY} = 1
\]

\[
P(a_1 < X \le a_2, b_1 < Y \le b_2) = \int_{a_1}^{a_2}\!\!\int_{b_1}^{b_2} f_{XY}(x,y)\,dy\,dx
\]

\textbf{Densités marginales} : on intègre l'autre variable.

\[
f_X(x) = \int_{-\infty}^\infty f_{XY}(x,y)\,dy \qquad f_Y(y) = \int_{-\infty}^\infty
f_{XY}(x,y)\,dx
\]

\end{formula}


\begin{formula}

\textbf{Cas discret}, $p_{ij} = P(X = x_i, Y = y_j)$ :

\[
f_{XY}(x,y) = \sum_{i,j} p_{ij}\,\delta(x-x_i)\,\delta(y-y_j)
\]

\end{formula}


\begin{definition}

$X$ et $Y$ sont \textbf{indépendantes} si $F_{XY}(x,y) = F_X(x)F_Y(y)$, d'où $f_{XY}(x,y) =
f_X(x)f_Y(y)$ (cas discret : $P(X = a, Y = b) = P(X=a)P(Y=b)$).

\vspace{3pt}

Si $X$ et $Y$ sont indépendantes, $g(X)$ et $h(Y)$ le sont aussi.

\end{definition}


\begin{simple}

La densité jointe, c'est le tas de sable posé \textbf{sur une table} (le plan $x,y$) au lieu
d'une ligne. La marginale de $X$, c'est ce qu'on voit en regardant le tas de côté : on écrase
tout le sable le long de $y$. Indépendantes : la forme du tas est juste « profil en $x$ fois
profil en $y$ », savoir $y$ ne dit rien sur $x$.

\end{simple}

\section{Fonctions de deux v.a. et jacobien}


\begin{formula}

$Z = g_1(X,Y)$, $W = g_2(X,Y)$, et $(x_i,y_i)$ les racines de $z = g_1(x,y)$, $w = g_2(x,y)$.
Jacobien :

\[
J(x,y) = \det\begin{pmatrix} \frac{\partial g_1}{\partial x} & \frac{\partial g_1}{\partial
y} \\[2pt] \frac{\partial g_2}{\partial x} & \frac{\partial g_2}{\partial y}\end{pmatrix}
\]

\[
f_{ZW}(z,w) = \sum_i \frac{f_{XY}(x_i,y_i)}{|J(x_i,y_i)|}
\]

\end{formula}


\begin{formula}

\textbf{Cas linéaire} $\mathbf z = A\mathbf x + \mathbf b$, avec $\det A \ne 0$ : une seule
racine $\mathbf x = A^{-1}(\mathbf z - \mathbf b)$, et $J = \det A$ partout.

\[
f_{\mathbf Z}(\mathbf z) = \frac{f_{\mathbf X}(A^{-1}(\mathbf z - \mathbf b))}{|\det A|}
\]

\end{formula}


\begin{example}

\textbf{Somme de deux v.a. indépendantes} $Z = X+Y$. Il faut deux équations : on ajoute $W =
Y$ (qui ne nous intéresse pas), ce qui donne $A = \begin{pmatrix}1&1\\0&1\end{pmatrix}$, $\det
A = 1$. Donc $f_{ZW}(z,w) = f_{XY}(z-w, w) = f_X(z-w)f_Y(w)$, puis la marginale :

\[
f_Z(z) = \int_{-\infty}^\infty f_X(z-w)\,f_Y(w)\,dw \quad\Rightarrow\quad f_Z = f_X \star f_Y
\]

\end{example}


\begin{important}

La densité de la \textbf{somme} de deux v.a. \textbf{indépendantes} est le \textbf{produit de
convolution} de leurs densités.

\end{important}


\begin{simple}

Comme pour une seule v.a. : on change de coordonnées, et le jacobien dit de combien la
transformation \textbf{étire ou écrase les surfaces}. Si elle étire, le sable s'étale et sa
hauteur baisse d'autant. Astuce à retenir : quand on veut une seule nouvelle variable (comme
$X+Y$), on invente une deuxième bidon ($W = Y$) pour pouvoir utiliser la formule, puis on
l'intègre pour s'en débarrasser.

\end{simple}

\section{Espérance, covariance et corrélation}


\begin{formula}

\[
E[g(X,Y)] = \iint g(x,y)\,f_{XY}(x,y)\,dx\,dy
\]

\begin{itemize}

\item $E[X+Y] = E[X] + E[Y]$ \textbf{toujours} (l'espérance est linéaire).

\item $E[XY] = E[X]E[Y]$ si $X$ et $Y$ sont \textbf{indépendantes}.

\item Moments joints : $E[X^nY^m] = \iint x^ny^m f_{XY}\,dx\,dy$.

\end{itemize}

\end{formula}


\begin{formula}

\textbf{Covariance} et \textbf{coefficient de corrélation} :

\[
\mathrm{COV}[X,Y] = E[(X-\mu_X)(Y-\mu_Y)] = E[XY] - E[X]E[Y]
\]
\[
\rho = \frac{\mathrm{COV}[X,Y]}{\sigma_X\sigma_Y}, \qquad \mathrm{COV}[X,X] = \sigma_X^2
\]

\begin{itemize}

\item $|\mathrm{COV}[X,Y]| \le \sigma_X\sigma_Y$, donc $|\rho| \le 1$ (exercice 7 : partir de
$E[(X-\lambda Y)^2] \ge 0$ pour tout $\lambda$).

\item \textbf{Décorrélées} : $\mathrm{COV}[X,Y] = 0$. \textbf{Orthogonales} : $E[XY] = 0$.

\end{itemize}

\end{formula}


\begin{warning}

Indépendantes $\Rightarrow$ décorrélées. \textbf{La réciproque est fausse} en général
(exercice 8 : $X = \cos\Theta$, $Y = \sin\Theta$ avec $\Theta$ uniforme sur $[0,2\pi]$).

\end{warning}


\begin{simple}

La covariance dit si deux nombres ont tendance à \textbf{monter ensemble} (positive), l'un
monte quand l'autre descend (négative), ou s'en fichent \textbf{en ligne droite} (nulle).
$\rho$, c'est la même chose ramenée entre $-1$ et $1$. Attention : $\rho = 0$ veut juste dire
« pas de lien en ligne droite ». Un point qui tourne sur un cercle a $X$ et $Y$ décorrélés,
pourtant si on connaît $X$, on connaît presque $Y$.

\end{simple}

\section{Transformées d'une paire}


\begin{formula}

\textbf{Fonction caractéristique jointe} :

\[
\Phi_{XY}(\omega_1,\omega_2) = E\big[e^{j(\omega_1X+\omega_2Y)}\big]
\]

(conjugué de la transformée de Fourier 2D de $f_{XY}$). Inverse :

\[
f_{XY}(x,y) = \frac{1}{4\pi^2}\iint e^{-j(\omega_1x+\omega_2y)}\Phi_{XY}(\omega_1,\omega_2)\,
d\omega_1 d\omega_2
\]

\begin{itemize}

\item Marginales : $\Phi_X(\omega) = \Phi_{XY}(\omega,0)$, $\Phi_Y(\omega) =
\Phi_{XY}(0,\omega)$.

\item Moments : $E[X^mY^n] = \frac{1}{j^{m+n}}\frac{\partial^{m+n}\Phi_{XY}}{\partial\omega_1^m
\partial\omega_2^n}\big|_{(0,0)}$.

\item Génératrice des moments : $\hat\Phi_{XY}(s_1,s_2) = E[e^{s_1X+s_2Y}]$.

\end{itemize}

\end{formula}


\begin{important}

Si $X$ et $Y$ sont \textbf{indépendantes} :

\[
\Phi_{XY}(\omega_1,\omega_2) = \Phi_X(\omega_1)\Phi_Y(\omega_2)
\]

et pour $Z = X+Y$ (convolution des densités $\leftrightarrow$ produit des transformées) :

\[
\Phi_Z(\omega) = \Phi_X(\omega)\,\Phi_Y(\omega)
\]

\end{important}


\begin{simple}

Additionner deux v.a. indépendantes, c'est une convolution, donc une intégrale pénible. Avec
les fonctions caractéristiques, c'est juste une \textbf{multiplication}. On multiplie, puis on
reconnaît la fonction caractéristique d'une loi connue.

\end{simple}

\section{Densité et espérance conditionnelles}


\begin{formula}

\textbf{Cas discret} : $P(X = x_i | Y = y_j) = \frac{P(X=x_i, Y=y_j)}{P(Y=y_j)}$ (0 si
$P(Y = y_j) = 0$).

\vspace{3pt}

\textbf{Cas continu} ($P(Y = y) = 0$, on passe par la limite sur $y \le Y < y+\Delta y$) :

\[
f_{X|Y}(x|y) = \frac{f_{XY}(x,y)}{f_Y(y)} \qquad f_{Y|X}(y|x) = \frac{f_{XY}(x,y)}{f_X(x)}
\]

\end{formula}


\begin{formula}

\textbf{Espérance conditionnelle} :

\[
E[X|y] = \int_{-\infty}^\infty x\,f_{X|Y}(x|y)\,dx
\]
\[
E[X|y_j] = \sum_i x_i\, P(X = x_i|Y=y_j)
\]

$E[X|y] = g(y)$ est une fonction \textbf{déterministe} de $y$, mais $E[X|Y] = g(Y)$ est une
\textbf{v.a.}, et :

\[
E\big[E[X|Y]\big] = E[X]
\]

\end{formula}


\begin{formula}

\textbf{Probabilités totales en espérance} ($[A_1,\dots,A_m]$ partition de $\Omega$) :

\[
E[g(X)] = \sum_{i=1}^m E[g(X)|A_i]\,P(A_i)
\]

$X$ discrète, $Y$ continue : $P(X = x_i) = \int P(X = x_i | Y = y) f_Y(y)\,dy$.

\end{formula}


\begin{simple}

$f_{X|Y}(x|y)$ : on \textbf{coupe} le tas de sable en tranche à la hauteur $Y = y$, et on
regarde le profil de cette tranche (renormalisé pour qu'il fasse 1). $E[X|Y]$ : « ma meilleure
moyenne pour $X$ si on me dit $Y$ ». Et si on fait la moyenne de ces moyennes sur tous les
$Y$ possibles, on retombe sur la moyenne de $X$ tout court.

\end{simple}

\section{V.a. complexes et vecteurs de dimension $n$}


\begin{formula}

$Z = X + jY$ ($X$, $Y$ réelles) :

\[
\mu_Z = \mu_X + j\mu_Y \qquad \sigma_Z^2 = E[|Z-\mu_Z|^2] = \sigma_X^2 + \sigma_Y^2
\]
\[
\mathrm{COV}[Z,W] = E[(Z^\star - \mu_Z^\star)(W-\mu_W)]
\]

Matrice de covariance hermitienne et semi-définie positive.

\end{formula}


\begin{formula}

\textbf{Vecteur} $\mathbf X = (X_1,\dots,X_n)$ :

\[
F_{\mathbf X}(\mathbf x) = P(X_1\le x_1,\dots,X_n\le x_n)
\]
\[
f_{\mathbf X}(\mathbf x) = \frac{\partial^n F_{\mathbf X}}{\partial x_1\cdots\partial x_n}
\]

Marginale : on intègre sur toutes les autres variables, $f_{X_1}(x_1) = \int\cdots\int
f_{\mathbf X}\,dx_2\cdots dx_n$.

\vspace{3pt}

Conditionnelle : $f_{X_1|X_2\dots X_n} = \frac{f_{X_1\dots X_n}}{f_{X_2\dots X_n}}$.

\end{formula}

\section{V.a. gaussiennes multivariées}


\begin{definition}

$X_1,\dots,X_n$ sont \textbf{gaussiennes jointes} (multivariées) ssi

\[
f_{\mathbf X}(\mathbf x) = \frac{1}{\sqrt{(2\pi)^n\det\Sigma}}\,e^{-\frac12(\mathbf x -
\boldsymbol\mu)^T\Sigma^{-1}(\mathbf x - \boldsymbol\mu)}
\]

avec $\boldsymbol\mu = [E[X_1]\dots E[X_n]]^T$ et la \textbf{matrice de covariance}
$\Sigma_{ik} = \mathrm{COV}[X_i,X_k]$ (variances sur la diagonale), symétrique et
semi-définie positive.

\end{definition}


\begin{formula}

\textbf{Fonction caractéristique} :

\[
\Phi_{\mathbf X}(\boldsymbol\omega) = e^{j\boldsymbol\mu^T\boldsymbol\omega - \frac12
\boldsymbol\omega^T\Sigma\boldsymbol\omega}
\]

\textbf{Moments} ($X_i$ gaussiennes jointes, de moyenne nulle) :

\[
E[X_1X_2X_3X_4] = E[X_1X_2]E[X_3X_4] + E[X_1X_3]E[X_2X_4]
\]
\[
+ E[X_1X_4]E[X_2X_3]
\]
\[
E[X_1X_2\cdots X_{2k+1}] = 0 \qquad E[X^{2k+1}] = 0
\]
\[
E[X^{2k}] = 1\cdot3\cdot5\cdots(2k-1)\,\sigma_X^{2k}
\]

\end{formula}


\begin{important}

\begin{itemize}

\item Une gaussienne jointe est \textbf{entièrement déterminée par ses moments d'ordre 1 et 2}
($\boldsymbol\mu$ et $\Sigma$) : tous les autres moments s'en déduisent.

\item Les courbes de niveau ($(\mathbf x-\boldsymbol\mu)^T\Sigma^{-1}(\mathbf x -
\boldsymbol\mu)$ constant) sont des \textbf{ellipses} en 2D, orientées selon $\sigma_{X_1}$,
$\sigma_{X_2}$ et $\rho$.

\end{itemize}

\end{important}


\begin{formula}

\textbf{Transformation linéaire} $\mathbf Y = A\mathbf X$ :

\[
f_{\mathbf Y}(\mathbf y) = \frac{1}{\sqrt{(2\pi)^n|\det S|}}\,e^{-\frac12(\mathbf y - \mathbf
m)^TS^{-1}(\mathbf y - \mathbf m)}
\]
\[
\text{avec}\quad S = A\Sigma A^T, \qquad \mathbf m = A\boldsymbol\mu
\]

Une transformation linéaire de gaussiennes jointes donne des \textbf{gaussiennes jointes}.

\end{formula}


\begin{important}

\textbf{Application} : on choisit $A$ qui \textbf{diagonalise} $\Sigma$, pour décorréler les
composantes du vecteur. (Le Moodle le rappelle pour l'exercice M3 10 : pour des gaussiennes
jointes, décorrélées veut dire indépendantes.)

\end{important}


\begin{simple}

Un vecteur gaussien, c'est une cloche en plusieurs dimensions. Vue de dessus, c'est une
\textbf{ellipse} : ronde si les composantes n'ont rien à voir, allongée et penchée si elles
sont liées. Super pouvoir : pour la connaître, il suffit de la \textbf{moyenne} (où est le
centre) et de la \textbf{covariance} (forme de l'ellipse). Et si on la tourne ou l'étire
(transformation linéaire), ça reste une cloche gaussienne : on peut la tourner pour que
l'ellipse soit bien droite, et alors les composantes deviennent indépendantes.

\end{simple}

\section{Rappels de traitement du signal (1) : systèmes LTI}


\begin{definition}

Résumé de Salim Najib (COM-202 pour COM-300), avec \textbf{les conventions de COM-300}.

\vspace{3pt}

\textbf{Signal} : une fonction $x:\mathbb R\to\mathbb C$ (continu) ou une suite $x:\mathbb
Z\to\mathbb C$ (discret). \textbf{Système} $\mathcal H$ : transforme un signal en un autre,
$y = \mathcal H(x)$.

\begin{itemize}

\item \textbf{Linéaire} : $\mathcal H(\alpha x_1 + x_2) = \alpha\mathcal H(x_1) + \mathcal
H(x_2)$.

\item \textbf{Invariant dans le temps} : décaler l'entrée de $t_0$ décale la sortie de $t_0$.

\end{itemize}

Linéaire + invariant = \textbf{LTI}, aussi appelé \textbf{filtre}.

\end{definition}


\begin{example}

$\mathcal H_1(x)(t) = x(t^2)$ est linéaire mais \textbf{pas} invariant : il change le
déroulement du temps. $\mathcal H_2(x)(n) = 0.5(x(n) + x(n-1))$ (moyenne des deux derniers
jours) est LTI.

\end{example}


\begin{formula}

\textbf{Réponse impulsionnelle} $h = \mathcal H(\delta)$ (Kronecker en discret, Dirac en
continu). Alors, pour tout signal $x$ :

\[
\mathcal H(x) = x * h
\]
\[
(x*h)(n) = \sum_{k\in\mathbb Z} x(k)h(n-k)
\]
\[
(x*h)(t) = \int_{\mathbb R} x(\tau)h(t-\tau)\,d\tau
\]

La convolution est bilinéaire et symétrique ($x*h = h*x$).

\end{formula}


\begin{formula}

\textbf{Causal} : $y(t)$ ne dépend que de $x(\tau)$ pour $\tau \le t$. \textbf{BIBO stable} :
entrée bornée $\Rightarrow$ sortie bornée. Pour un système LTI :

\begin{itemize}

\item causal $\Leftrightarrow h(t) = 0$ pour $t < 0$ ;

\item BIBO stable $\Leftrightarrow \sum_n|h(n)| < \infty$ (discret), $\int|h(t)|\,dt <
\infty$ (continu).

\end{itemize}

\end{formula}


\begin{simple}

Un filtre LTI, c'est une \textbf{machine qui ne change pas d'humeur} (invariante) et qui
traite les morceaux d'un signal séparément puis additionne (linéaire). Pour savoir tout ce
qu'elle fait, il suffit de lui donner \textbf{une seule claque} (l'impulsion $\delta$) et
d'écouter sa réponse $h$. N'importe quel signal est une somme de petites claques, donc la
sortie est la somme des petites réponses : c'est la convolution.

\end{simple}

\section{Rappels de traitement du signal (2) : transformées}


\begin{formula}

\textbf{Fourier (convention COM-300)} :

\[
X(f) = \int_{\mathbb R} x(t)\,e^{-j2\pi ft}\,dt
\]

\begin{itemize}

\item $\mathcal F(\delta) = 1$ ; $\mathcal F(x(\cdot - t_0))(f) = e^{-j2\pi ft_0}X(f)$ ;
$\mathcal F(x')(f) = j2\pi f X(f)$ ;

\item $\mathcal F(x * y) = \mathcal F(x)\,\mathcal F(y)$ ; inverse : $\mathcal F^{-1}(X)(t) =
\mathcal F(X)(-t)$.

\end{itemize}

\textbf{Réponse fréquentielle} $H = \mathcal F(h)$ : $y = h*x \Rightarrow Y(f) = H(f)X(f)$.

\end{formula}


\begin{example}

$h(t) = \operatorname{sinc}(t) = \frac{\sin\pi t}{\pi t}$, $H(f) = \operatorname{rect}(f) =
\mathbb 1[|f| < \frac12]$, et $x(t) = \operatorname{sinc}(t/2)$, $X(f) = 2\cdot\mathbb 1[|f| <
\frac14]$. La convolution est pénible, mais $Y(f) = H(f)X(f) = 2\cdot\mathbb 1[|f|<\frac14] =
X(f)$ : donc $y = x$.

\end{example}


\begin{formula}

\textbf{Transformée en $Z$} (signal discret) :

\[
\hat x(z) = \sum_{n\in\mathbb Z} x(n)z^{-n}, \quad z \in \mathrm{ROC}
\]

Linéaire, $\mathcal Z(x*y) = \mathcal Z(x)\mathcal Z(y)$, décalage : $x(n-n_0) \to
z^{-n_0}\hat x(z)$.

\vspace{3pt}

\textbf{Fourier discrète} (DTFT), si le cercle unité est dans la ROC :

\[
X(f) = \hat x(e^{j2\pi f}) = \sum_{n\in\mathbb Z} x(n)e^{-j2\pi fn}
\]

\textbf{Fonction de transfert} $\hat h = \mathcal Z(h)$ : $\hat y(z) = \hat h(z)\hat x(z)$.

\end{formula}


\begin{example}

\textbf{Équation de récurrence} $y(n) + a\,y(n-1) = x(n) + b\,x(n-1) + c\,x(n-2)$. On applique
$\mathcal Z$ des deux côtés :

\[
\hat y(z)(1 + az^{-1}) = \hat x(z)(1 + bz^{-1} + cz^{-2})
\]
\[
\hat h(z) = \frac{1 + bz^{-1} + cz^{-2}}{1 + az^{-1}}
\]

\end{example}


\begin{formula}

\textbf{Laplace} (celle d'Analyse IV, utilisée en COM-300 pour des signaux \textbf{causaux}) :

\[
\mathcal L(x)(s) = \int_0^\infty x(t)e^{-st}\,dt, \qquad \mathcal L(x')(s) = s\,\mathcal L(x)(s)
- x(0)
\]

\end{formula}


\begin{warning}

La convention de Fourier de COM-300 n'est \textbf{pas} celle de COM-202 (ni du livre de
Prandoni et Vetterli). Et attention : $\Phi_X(\omega) = E[e^{j\omega X}]$ a un signe $+$ dans
l'exponentielle, c'est le \textbf{conjugué} de la transformée de Fourier de $f_X$.

\end{warning}


\begin{simple}

Une transformée, c'est regarder le signal en \textbf{notes de musique} (fréquences) au lieu
de le regarder dans le temps. L'avantage : la convolution, compliquée dans le temps, devient
une simple \textbf{multiplication} en fréquence. Le filtre devient « je multiplie chaque note
par un réglage du volume $H(f)$ ».

\end{simple}
