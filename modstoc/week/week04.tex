\begin{important}

\textbf{Conseil du Moodle pour 4 g} : les moments d'ordre supérieur de gaussiennes multivariées
s'expriment en fonction des moments d'ordre 1 et 2 (par exemple $E[X_1X_2X_3X_4]$, semaine 2).

\end{important}


\begin{simple}

Jusqu'ici : des nombres au hasard. Maintenant : des \textbf{courbes} au hasard. Un processus,
c'est une courbe qui dépend du temps, et dont la forme est tirée au sort (le bruit qu'on entend
dans une radio, la tension d'un fil bruité). On va surtout les décrire par leur moyenne et par
la façon dont deux instants sont liés.

\end{simple}

\section{Processus stochastique : définitions}


\begin{definition}

Un \textbf{processus stochastique} $X(t,\zeta)$ est une fonction à la fois du résultat $\zeta$
et du temps $t$. Temps continu ($t\in\mathbb R$) dans ce module.

\begin{itemize}

\item $\zeta$ fixé : une fonction déterministe du temps, une \textbf{trajectoire}
(réalisation).

\item $t$ fixé : une \textbf{v.a.} $X(t)$.

\end{itemize}

En échantillonnant en $t_1,\dots,t_n$, on obtient $n$ v.a. Le processus est spécifié par toutes
les répartitions jointes, pour tout $n$ et tous $t_1,\dots,t_n$ :

\[
F_{X_1\dots X_n}(x_1,\dots,x_n;t_1,\dots,t_n)
\]
\[
= P(X(t_1)\le x_1;\dots;X(t_n)\le x_n)
\]

\end{definition}


\begin{formula}

\textbf{Analyse au second ordre} : on se contente des moments d'ordre 1 et 2.

\[
\mu_X(t) = E[X(t)]
\]
\[
R_X(t_1,t_2) = E[X(t_1)X(t_2)] \quad\text{(auto-corrélation)}
\]
\[
C_X(t_1,t_2) = R_X(t_1,t_2) - \mu_X(t_1)\mu_X(t_2)
\]

(auto-covariance), et le coefficient d'auto-corrélation :
\[
\rho_X(t_1,t_2) = \frac{C_X(t_1,t_2)}{\sqrt{C_X(t_1,t_1)C_X(t_2,t_2)}}
\]

$C_X(t,t) = \mathrm{VAR}[X(t)]$ ; $R_X(t,t) = E[X^2(t)]$ : la \textbf{puissance moyenne}.

\end{formula}


\begin{formula}

\textbf{Deux processus} : cross-corrélation et cross-covariance

\[
R_{XY}(t_1,t_2) = E[X(t_1)Y(t_2)]
\]
\[
C_{XY}(t_1,t_2) = R_{XY}(t_1,t_2) - \mu_X(t_1)\mu_Y(t_2)
\]

\textbf{Orthogonaux} : $R_{XY} = 0$ pour tous $t_1,t_2$. \textbf{Non corrélés} : $C_{XY} = 0$
pour tous $t_1,t_2$.

\end{formula}


\begin{simple}

Imaginez 1000 personnes qui enregistrent chacune le bruit de leur radio : 1000 courbes
différentes (les trajectoires). Si on regarde \textbf{toutes} les courbes à 14h pile, on a 1000
nombres : c'est la v.a. $X(t)$. $\mu_X(t)$ : la moyenne de ces nombres à l'instant $t$.
$R_X(t_1,t_2)$ : est-ce que la valeur à $t_1$ et celle à $t_2$ ont tendance à aller ensemble ?

\end{simple}

\section{Exemples : sinusoïdes aléatoires}


\begin{example}

\textbf{Sinusoïde à phase aléatoire} $X(t) = a\sin(2\pi f_0t + \Phi)$, $a$, $f_0$ constantes.

\vspace{3pt}

\textbf{Cas 1 : $\Phi \in \{0,\pi\}$ équiprobable} (Bernoulli). Deux trajectoires,
$a\sin(2\pi f_0t)$ et $-a\sin(2\pi f_0t)$. À $t_0$ fixé, $X(t_0) = \pm a\sin(2\pi f_0 t_0)$.

\[
\mu_X(t) = 0
\]
\[
R_X(t_1,t_2) = a^2\sin(2\pi f_0t_1)\sin(2\pi f_0t_2)
\]
\[
= \frac{a^2}2\big[\cos(2\pi f_0(t_1-t_2)) - \cos(2\pi f_0(t_1+t_2))\big]
\]

\textbf{Cas 2 : $\Phi$ uniforme sur $[0,2\pi]$} ($X(t)$ est une v.a. continue ; exercice 1) :

\[
\mu_X(t) = 0 \qquad R_X(t_1,t_2) = \frac{a^2}2\cos(2\pi f_0(t_1-t_2))
\]

\end{example}


\begin{example}

\textbf{Sinusoïdes en quadrature à amplitudes aléatoires} : $X(t) = A\cos(2\pi f_0 t) +
B\sin(2\pi f_0t)$, $A$ et $B$ indépendantes, moyennes nulles, même variance $\sigma^2$.

\[
\mu_X(t) = 0 \qquad R_X(t_1,t_2) = \sigma^2\cos(2\pi f_0(t_1-t_2))
\]

Si $\sigma^2 = a^2/2$, mêmes moments d'ordre 1 et 2 que la sinusoïde à phase uniforme, mais
\textbf{en général ce sont deux processus différents}.

\end{example}


\begin{simple}

On a une onde parfaite, mais on ne sait pas \textbf{où elle commence}. Si le départ peut être
n'importe où (phase uniforme), toutes les heures se ressemblent : le lien entre deux instants
ne dépend que de l'écart $t_1 - t_2$. Si la phase ne vaut que 0 ou $\pi$, certaines heures
sont spéciales (là où le sinus vaut 0, $X$ vaut 0 à coup sûr), et $R_X$ dépend aussi de $t_1 +
t_2$.

\end{simple}

\section{Processus particuliers}


\begin{definition}

\textbf{Gaussien} : $X(t_1),\dots,X(t_n)$ gaussiennes jointes pour tout $n$ et tous les $t_i$,
de matrice de covariance $(\Gamma_X)_{ik} = C_X(t_i,t_k)$. \textbf{Entièrement déterminé par
$\mu_X$ et $C_X$} : l'analyse au second ordre le caractérise complètement. Si $X(t)$ est
gaussien, $\int X(t)\,dt$ l'est aussi.

\vspace{3pt}

\textbf{Markovien} : pour tous $t_1<\dots<t_{k+1}$,

\[
P(X(t_{k+1})\le x_{k+1} | X(t_k) = x_k,\dots,X(t_1) = x_1)
\]
\[
= P(X(t_{k+1})\le x_{k+1}|X(t_k) = x_k)
\]

\textbf{À accroissements indépendants} : pour tous $t_1<\dots<t_k$, les v.a. $X(t_k) -
X(t_{k-1}), \dots, X(t_2) - X(t_1)$ sont indépendantes.

\end{definition}


\begin{definition}

\textbf{Wiener} (mouvement brownien, fin du cours) : accroissements indépendants et $X(t_2) -
X(t_1)\sim\mathcal N(0, t_2-t_1)$. Gaussien, et :

\[
\mu_X(t) = 0 \qquad R_X(t_1,t_2) = \min\{t_1,t_2\}
\]

\textbf{Poisson homogène} de paramètre $\lambda$ (module 5) : accroissements indépendants, et
le nombre d'évènements dans $[t_1,t_2]$ suit une loi de Poisson de paramètre
$\lambda(t_2-t_1)$.

\[
\mu_X(t) = \lambda t \qquad R_X(t_1,t_2) = \lambda^2t_1t_2 + \lambda\min\{t_1,t_2\}
\]

\textbf{Martingale} : pour $t_1 < t_2$, $E[X(t_2)|X(s), s\le t_1] = X(t_1)$ (le processus de
Wiener en est une).

\end{definition}


\begin{simple}

\begin{itemize}

\item \textbf{Markov} : pour deviner la suite, seul le \textbf{présent} compte, le passé ne
sert à rien de plus. Un poisson rouge qui n'a pas de mémoire.

\item \textbf{Accroissements indépendants} : ce qui se passe dans un intervalle de temps n'a
rien à voir avec ce qui s'est passé dans un autre.

\item \textbf{Martingale} : un jeu équitable, en moyenne votre fortune de demain est celle
d'aujourd'hui.

\end{itemize}

\end{simple}

\section{Train de pulses modulés en amplitude (PAM)}


\begin{definition}

Pulses $g(t-kT)$ modulés par des v.a. $A_k$ :

\[
X_c(t) = \sum_{k=-\infty}^{\infty}A_k\,g(t-kT)
\]

avec $\int|g| < \infty$, $\int|g|^2 < \infty$, $E[A_k] = \mu_A$, $E[A_kA_l] = R_A(k-l)$. Sa
moyenne $\mu_{X_c}(t) = \mu_A\sum_kg(t-kT)$ dépend de $t$.

\vspace{3pt}

On ajoute un \textbf{décalage aléatoire} $D$ uniforme sur $[0,T]$, indépendant des $A_k$ :

\[
X(t) = \sum_{k=-\infty}^{\infty}A_k\,g(t-kT-D)
\]

\end{definition}


\begin{formula}

\[
\mu_X(t) = \frac{\mu_A}{T}\int_{-\infty}^{\infty}g(s)\,ds
\]
\[
R_X(t_1,t_2) = \frac1T\sum_{n=-\infty}^{\infty}R_A(n)\int_{-\infty}^{\infty}g(s)\,g(t_2-t_1+nT+s)
\,ds
\]

\end{formula}


\begin{formula}

\textbf{Séquence binaire aléatoire} : pulse rectangulaire $g(t) = a\,\mathbb 1_{[0,T]}(t)$ et
$A_k$ \textbf{indépendants} ($R_A(0) = \sigma_A^2 + \mu_A^2$, $R_A(n) = \mu_A^2$ sinon) :

\[
\mu_X = a\mu_A
\]
\[
R_X(t_1,t_2) = \begin{cases} a^2\sigma_A^2\big(1 - \frac{|t_1-t_2|}{T}\big) + a^2\mu_A^2 &
|t_1-t_2| < T \\ a^2\mu_A^2 & \text{sinon}\end{cases}
\]
\[
C_X(t_1,t_2) = \begin{cases} a^2\sigma_A^2\big(1 - \frac{|t_1-t_2|}{T}\big) & |t_1-t_2| < T \\
0 & \text{sinon}\end{cases}
\]

(Exercice 2 : le retrouver plus simplement en conditionnant sur $D$, avec les probabilités
totales.)

\end{formula}


\begin{simple}

C'est la façon d'envoyer des bits : toutes les $T$ secondes, on envoie une petite impulsion
$g$, multipliée par le symbole $A_k$ (par exemple $+1$ ou $-1$). Le décalage $D$, c'est qu'on
ne sait pas quand le premier symbole a commencé : ça rend toutes les heures équivalentes. Le
résultat : deux instants dans le \textbf{même} créneau sont liés (même symbole), deux instants
écartés de plus de $T$ ne le sont plus (symboles indépendants), d'où le triangle.

\end{simple}
