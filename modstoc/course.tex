\section{Bases des probabilités}

\begin{rembox}[title=Contexte]
La théorie des probabilités est l'ensemble des techniques mathématiques décrivant les
résultats d'expériences dont l'issue n'est pas connue à l'avance.
\end{rembox}

\begin{defnbox}[title=Notations de base]
Soient $\zeta_i$ le résultat de la $i$-ème expérience, $A, B, \ldots = \{\text{quelques }
\zeta_i\}$ un \textbf{évènement}, $\Omega = \{\text{tous les } \zeta_i\}$ l'\textbf{évènement
certain} et $\emptyset$ l'\textbf{évènement impossible}.
\end{defnbox}

\subsection{Axiomes de probabilité}

\begin{propbox}[title=Axiomes de probabilité]
La probabilité de l'évènement $A$ est le réel $P(A)$ satisfaisant aux axiomes suivants :
\begin{itemize}
  \item[A1.] $P(A) \geq 0$
  \item[A2.] $P(\Omega) = 1$
  \item[A3.] Dans le cas où $\Omega$ est un ensemble fini, si $A \cap B = \emptyset$ alors
        $P(A \cup B) = P(A) + P(B)$.
  \item[A3'.] Dans le cas où $\Omega$ est un ensemble infini, si $A_1, A_2, \ldots$ est une
        suite d'évènements telle que $A_i \cap A_j = \emptyset$ pour $i \neq j$ alors
        $P\!\left(\bigcup_{i=1}^{\infty} A_i\right) = \sum_{i=1}^{\infty} P(A_i)$.
\end{itemize}
\end{propbox}

\begin{propbox}[title=Corollaires des axiomes]
Ces axiomes entraînent les corollaires suivants :
\begin{itemize}
  \item[P1.] $P(\bar{A}) = 1 - P(A)$ où $\bar{A} = \Omega \setminus A$.
  \item[P2.] $P(A) \leq 1$.
  \item[P3.] $P(\emptyset) = 0$.
  \item[P4.] $P(A \cup B) = P(A) + P(B) - P(A \cap B)$.
  \item[P5.] Si $A \subseteq B$, $P(A) \leq P(B)$.
\end{itemize}
\end{propbox}

\subsection{Probabilité conditionnelle}

\begin{defnbox}[title=Probabilité conditionnelle]
La probabilité de l'évènement $A$ étant donné que l'évènement $B$ s'est déroulé est définie
par
\[
P(A \mid B) = \frac{P(A \cap B)}{P(B)}
\]
si $P(B) \neq 0$ et par $P(A \mid B) = 0$ si $P(B) = 0$.
\end{defnbox}

\begin{defnbox}[title=Partition]
$[A_1, A_2, \ldots, A_n]$ est une \textbf{partition} de $\Omega$, c'est-à-dire un ensemble
d'évènements tels que
\[
A_1 \cup \ldots \cup A_n = \Omega \qquad \text{et} \qquad A_i \cap A_j = \emptyset
\text{ pour } i \neq j.
\]
\end{defnbox}

\begin{propbox}[title=Théorème des probabilités totales -- P6]
Soit $[A_1, \ldots, A_n]$ une partition de $\Omega$. Alors
\[
P(B) = \sum_{i=1}^{n} P(B \mid A_i) P(A_i).
\]
\end{propbox}

\begin{propbox}[title=Règle de Bayes -- P7]
\[
P(A_i \mid B) = \frac{P(B \mid A_i) P(A_i)}{\sum_{j=1}^{n} P(B \mid A_j) P(A_j)}.
\]
\end{propbox}

\subsection{Indépendance}

\begin{defnbox}[title=Indépendance de deux évènements]
Deux évènements $A$ et $B$ sont \textbf{indépendants} lorsque
\[
P(A \cap B) = P(A) P(B).
\]
\end{defnbox}

\begin{defnbox}[title=Indépendance de trois évènements]
Trois évènements $A$, $B$ et $C$ sont indépendants si et seulement si
\begin{align*}
P(A \cap B) &= P(A) P(B) \\
P(A \cap C) &= P(A) P(C) \\
P(B \cap C) &= P(B) P(C)
\end{align*}
et
\[
P(A \cap B \cap C) = P(A) P(B) P(C).
\]
On généralise aisément cette définition à un nombre quelconque d'évènements
indépendants.
\end{defnbox}

\section{Variable aléatoire}

\begin{defnbox}[title=Variable aléatoire (v.a.)]
Une \textbf{variable aléatoire} (v.a.) est une fonction qui assigne à chaque résultat $\zeta$
d'une expérience un réel $X(\zeta)$ tel que
\begin{itemize}
  \item[C1.] l'ensemble $\{\zeta \mid X(\zeta) \leq x\}$ est un évènement pour tout $x \in
        \mathbb{R}$ ;
  \item[C2.] les probabilités des évènements $\{\zeta \mid X(\zeta) = \infty\}$ et $\{\zeta
        \mid X(\zeta) = -\infty\}$ sont nulles, i.e. $P(X = \infty) = P(X = -\infty) = 0$.
\end{itemize}
\end{defnbox}

\subsection{Fonction de répartition}

\begin{defnbox}[title=Fonction de répartition]
Une v.a. est caractérisée par sa \textbf{fonction de répartition} $F_X(x)$, qui est la
probabilité que $X$ prenne une valeur inférieure ou égale au réel $x$ :
\[
F_X(x) = P(X \leq x) = P(A), \qquad A = \{\zeta \in \Omega \mid X(\zeta) \leq x\}.
\]
\end{defnbox}

\begin{propbox}[title=Propriétés de $F_X(x)$]
\begin{itemize}
  \item[P8.] $0 \leq F_X(x) \leq 1$.
  \item[P9.] $\lim_{x \to -\infty} F_X(x) = 0$ et $\lim_{x \to \infty} F_X(x) = 1$.
  \item[P10.] Si $a < b$, $F_X(a) \leq F_X(b)$.
  \item[P11.] $P(a < X \leq b) = F_X(b) - F_X(a)$.
  \item[P12.] $F_X(x)$ est continue à droite, i.e. $F_X(x) = \lim_{\eps \to 0, \eps > 0}
        F_X(x + \eps) = F_X(x^+)$.
\end{itemize}
\end{propbox}

\begin{defnbox}[title=Types de variables aléatoires]
On note $S_X$ l'ensemble des valeurs que peut prendre $X$ : $S_X = \{X(\zeta) \mid \zeta \in
\Omega\}$.
\begin{itemize}
  \item Si $X$ prend un ensemble dénombrable de valeurs $S_X = \{x_1, x_2, x_3, \ldots\}$,
        $X$ est appelée \textbf{v.a. discrète}. Sa fonction de répartition est une fonction
        en escalier, avec une discontinuité à gauche en chaque valeur $x_i$ :
        \[
        p_i = P(X = x_i) = F_X(x_i) - \lim_{\eps \to 0, \eps > 0} F_X(x_i - \eps) =
        F_X(x_i) - F_X(x_i^-).
        \]
  \item Si $F_X(x)$ est continue, on parle de \textbf{v.a. continue}. Dans ce cas,
        $P(X = x) = 0$ pour tout $x$ particulier.
  \item Dans les autres cas, $F(x)$ est discontinue bien que n'étant pas une fonction en
        escalier : on a alors une \textbf{v.a. mixte}.
\end{itemize}
\end{defnbox}

\subsection{Densité de probabilité}

\begin{defnbox}[title=Densité de probabilité]
La fonction de \textbf{densité de probabilité} est définie par
\[
f_X(x) = \frac{dF_X(x)}{dx}.
\]
\end{defnbox}

\begin{propbox}[title=Propriétés de $f_X(x)$]
\begin{itemize}
  \item[P13.] $f_X(x) \geq 0$.
  \item[P14.] $\int_{-\infty}^{\infty} f_X(x)\,dx = 1$.
  \item[P15.] $P(X \leq a) = F_X(a) = \int_{-\infty}^{a} f_X(x)\,dx$.
  \item[P16.] $P(a < X \leq b) = F_X(b) - F_X(a) = \int_a^b f_X(x)\,dx$. Cette propriété
        entraîne que $P(x < X \leq x + \Delta x) = f_X(x)\Delta x$ lorsque $\Delta x \to 0$.
\end{itemize}
\end{propbox}

\begin{rembox}[title=Cas discret -- impulsions de Dirac]
Dans le cas d'une v.a. discrète, l'utilisation des impulsions de Dirac $\delta(\cdot)$
permet d'écrire la densité sous la forme
\[
f_X(x) = \sum_i p_i \, \delta(x - x_i)
\]
où $p_i = P(X = x_i)$. Les propriétés P13 à P16 deviennent alors
\begin{itemize}
  \item[P13'.] $p_i \geq 0$.
  \item[P14'.] $\sum_i p_i = 1$.
  \item[P15'.] $P(X \leq a) = F_X(a) = \sum_{x_i \leq a} p_i$.
  \item[P16'.] $P(X = x_i) = F_X(x_i) - F_X(x_i^-) = p_i$.
\end{itemize}
\end{rembox}

\subsection{Fonction d'une variable aléatoire}

Soit $g(\cdot)$ une fonction continue. Si $X$ est une v.a. continue, l'expression
$Y = g(X)$ désigne également une v.a. continue. Soit $y$ une valeur particulière de $Y$, et
$x_1, x_2, \ldots, x_m$ les $m$ racines respectives de l'équation $y = g(x)$.

\begin{propbox}[title={Densité de $Y = g(X)$}]
La densité de probabilité de $Y = g(X)$ est donnée par
\[
f_Y(y) = \sum_i \frac{f_X(x_i)}{|g'(x_i)|}
\]
où la somme porte sur toutes les racines $x_i$ de $y = g(x)$. Cette formule est tout à fait
générale, et s'applique à tous les cas où $X$ est continue.
\end{propbox}

\subsection{Espérance d'une fonction d'une variable aléatoire}

\begin{defnbox}[title=Espérance]
L'\textbf{espérance} de la fonction $g(X)$, où $X$ est une v.a. de densité de probabilité
$f_X(x)$, est définie par
\[
\E[g(X)] = \int_{-\infty}^{\infty} g(x) f_X(x)\,dx
\]
pour autant que cette intégrale converge absolument.
\end{defnbox}

\begin{defnbox}[title=Moment d'ordre $n$]
Si $g(X) = X^n$, l'espérance porte le nom de \textbf{moment d'ordre $n$} :
\[
\E[X^n] = \int_{-\infty}^{\infty} x^n f_X(x)\,dx.
\]
Le moment d'ordre 1 est encore appelé \textbf{moyenne} de la v.a. $X$ et est noté
\[
\mu_X = \E[X] = \int_{-\infty}^{\infty} x f_X(x)\,dx.
\]
\end{defnbox}

\begin{defnbox}[title=Variance et écart-type]
La v.a. $Y = X - \mu_X$ est une v.a. \textbf{centrée} vu que $\E[Y] = 0$. La
\textbf{variance} de la v.a. $X$ est le moment d'ordre 2 de $Y = X - \mu_X$, et l'écart-type
$\sigma_X$ est la racine carrée de la variance :
\begin{align*}
\Var[X] = \sigma_X^2 &= \E[(X - \mu_X)^2] = \int_{-\infty}^{\infty} (x - \mu_X)^2 f_X(x)\,dx
\\
&= \E[X^2] - \mu_X^2 = \int_{-\infty}^{\infty} x^2 f_X(x)\,dx - \mu_X^2.
\end{align*}
Dans le cas discret, avec $P(X = x_i) = p_i$,
\[
\mu_X = \E[X] = \sum_i p_i x_i, \qquad \Var[X] = \sigma_X^2 = \E[(X-\mu_X)^2] = \sum_i p_i
(x_i - \mu_X)^2.
\]
\end{defnbox}

\begin{exbox}[title=Une v.a. sans moments -- la loi de Cauchy]
Les moments peuvent ne pas exister pour certaines v.a. La densité de probabilité de la v.a.
de Cauchy est
\[
f_X(x) = \frac{1}{\pi (1 + x^2)}.
\]
On vérifie qu'il s'agit bien d'une densité de probabilité :
\[
\int_{-\infty}^{\infty} f_X(x)\,dx = \frac{1}{\pi}\Big[\arctan(x)\Big]_{-\infty}^{\infty} = 1.
\]
Par contre,
\[
\int_{-\infty}^{\infty} |x f_X(x)|\,dx = \frac{2}{\pi} \int_0^{\infty} \frac{x}{1+x^2}\,dx =
\frac{1}{\pi}\Big[\ln(1+x^2)\Big]_0^{\infty} = \infty,
\]
donc $x f_X(x)$ n'est pas absolument intégrable, et $\E[X]$ n'est pas défini.
\end{exbox}

\subsection{Transformées}

\begin{defnbox}[title=Fonction caractéristique]
L'opérateur d'espérance mathématique appliqué à la v.a. complexe $\exp(j\omega X)$, où $j$
est la racine carrée de $-1$, définit la \textbf{fonction caractéristique} $\Phi_X(\omega)$
de la v.a. $X$ :
\[
\Phi_X(\omega) = \E[e^{j\omega X}] = \int_{-\infty}^{\infty} e^{j\omega x} f_X(x)\,dx.
\]
C'est le complexe conjugué de la transformée de Fourier de $f_X(x)$. Elle existe toujours car
$|\Phi_X(\omega)| \leq \int_{-\infty}^{\infty} |f_X(x)|\,dx = 1$, et est maximale en $\omega =
0$ car $\Phi_X(0) = 1$.
\end{defnbox}

\begin{propbox}[title=Transformée inverse]
\[
f_X(x) = \frac{1}{2\pi} \int_{-\infty}^{\infty} e^{-j\omega x} \Phi_X(\omega)\,d\omega.
\]
Si deux v.a. continues ont la même fonction caractéristique, elles ont la même densité de
probabilité, et vice-versa.
\end{propbox}

\begin{propbox}[title=Moments via la fonction caractéristique]
En dérivant $\Phi_X(\omega)$ par rapport à $\omega$ et en évaluant en $\omega = 0$ :
\[
\frac{d\Phi_X(\omega)}{d\omega}\bigg|_{\omega=0} = j \E[X] \quad \Longrightarrow \quad
\E[X] = \frac{1}{j} \frac{d\Phi_X(\omega)}{d\omega}\bigg|_{\omega=0}.
\]
Plus généralement, pour le moment d'ordre $k$ :
\[
\E[X^k] = \frac{1}{j^k} \frac{d^k \Phi_X(\omega)}{d\omega^k}\bigg|_{\omega=0}.
\]
\end{propbox}

\begin{defnbox}[title=Fonction génératrice des moments]
De la même manière qu'on peut étendre la transformée de Fourier à la transformée de Laplace
en prenant un complexe $s$ au lieu de $j\omega$, on définit la \textbf{fonction génératrice
des moments} :
\[
\hat{\Phi}_X(s) = \E[e^{sX}] = \int_{-\infty}^{\infty} e^{sx} f_X(x)\,dx.
\]
\end{defnbox}

\begin{defnbox}[title=Fonction génératrice de cumulants]
On utilise aussi parfois la \textbf{fonction génératrice de cumulant}, ou \textbf{seconde
fonction caractéristique},
\[
\Psi_X(\omega) = \ln \Phi_X(\omega).
\]
\end{defnbox}

\begin{defnbox}[title=Fonction génératrice de probabilité (cas discret)]
Dans le cas d'une v.a. discrète $X$ prenant des valeurs uniformément espacées $S_X = \{0, 1,
2, 3, \ldots\} = \mathbb{N}$, on définit une \textbf{fonction génératrice de probabilité},
qui est l'espérance de la fonction $z^X$, i.e. la transformée en $z$ de $p_k$ :
\[
G_X(z) = \E[z^X] = \sum_{k=0}^{\infty} z^k P(X=k) = \sum_{k=0}^{\infty} z^k p_k.
\]
\end{defnbox}

\begin{propbox}[title=Propriétés de $G_X(z)$]
On retrouve les probabilités $p_k$ en dérivant $k$ fois $G_X(z)$ par rapport à $z$, et en
évaluant cette dérivée en $z = 0$ :
\[
p_k = P(X=k) = \frac{1}{k!} \frac{d^k G_X(z)}{dz^k}\bigg|_{z=0}.
\]
En dérivant $k$ fois $G_X(z)$ mais en évaluant cette dérivée à présent en $z=1$, on trouve
les moments factoriaux de $X$, à partir desquels on peut retrouver tous les moments de $X$ :
\[
\E[X(X-1)(X-2)\cdots(X-k+1)] = \frac{d^k G_X(z)}{dz^k}\bigg|_{z=1}.
\]
Remarquons finalement que $G_X(1) = \sum_{k=0}^{\infty} p_k = 1$.
\end{propbox}

\subsection{Inégalités}

\begin{rembox}[title=Pourquoi borner plutôt que calculer ?]
Très souvent en pratique, on ne connaît pas complètement la densité de probabilité $f_X(x)$
d'une v.a. $X$. Il est donc très utile de pouvoir borner une probabilité ou l'espérance d'une
fonction de cette v.a.
\end{rembox}

\begin{propbox}[title=Inégalité de Markov]
Si $X \geq 0$ et $a > 0$,
\[
P(X \geq a) \leq \frac{\E[X]}{a}.
\]
\end{propbox}

\begin{propbox}[title=Inégalité de Tchebytcheff]
En posant $D = (X-\mu_X)^2$ et $b = \sqrt{a}$ dans l'inégalité de Markov,
\[
P(|X - \mu_X| \geq b) \leq \frac{\sigma_X^2}{b^2}.
\]
\end{propbox}

\begin{thmbox}[title=Inégalité de Jensen]
Une fonction $g : I \subseteq \mathbb{R} \to \mathbb{R} \cup \{\infty\}$ est \textbf{convexe}
si pour tout $x, x' \in I$ et pour tout $0 \leq t \leq 1$,
\[
g(tx + (1-t)x') \leq t\,g(x) + (1-t)\,g(x').
\]
Une fonction $g$ convexe et continue est toujours minorée par une fonction linéaire passant
par un point de son graphe : pour tout $a \in \mathbb{R}$, il existe $\theta \in \mathbb{R}$
tel que pour tout $x \in \mathbb{R}$, $g(x) \geq g(a) + \theta(x-a)$. En prenant $a = \E[X]$
et $x = X$, puis en prenant l'espérance des deux côtés, il s'ensuit que
\[
\E[g(X)] \geq g(\E[X]).
\]
\end{thmbox}

\section{Exemples de variables aléatoires}

\begin{rembox}[title=Panorama]
\begin{itemize}
  \item \textbf{V.a. discrètes} : Bernoulli, binomiale, géométrique, Poisson.
  \item \textbf{V.a. continues} : uniforme, gaussienne, exponentielle, Gamma, Erlang,
        Chi-carré.
\end{itemize}
\end{rembox}

\subsection{Variable aléatoire de Bernoulli}

\begin{defnbox}[title=Fonction indicatrice]
La \textbf{fonction indicatrice} $I_A$ d'un évènement $A$ lié à une expérience aléatoire est
\[
I_A(\zeta) = \begin{cases} 0 & \text{si } \zeta \notin A \\ 1 & \text{si } \zeta \in A.
\end{cases}
\]
\end{defnbox}

\begin{defnbox}[title=V.a. de Bernoulli]
La fonction indicatrice $X = I_A$ est une v.a. qui prend une des valeurs de $S_X = \{0,1\}$
avec probabilités $P(X=0) = 1-p$ et $P(X=1) = p$, où $p = P(A)$, et est appelée
\textbf{v.a. de Bernoulli} de paramètre $p$.
\[
\mu_X = \E[X] = p, \qquad \sigma_X^2 = p(1-p), \qquad G_X(z) = 1 - p + pz.
\]
\end{defnbox}

\subsection{Variable aléatoire binomiale}

\begin{defnbox}[title=V.a. binomiale]
On répète une expérience $A$ de Bernoulli, de paramètre $p$, $n$ fois indépendamment. La v.a.
$X$ comptant le nombre de succès obtenus est la \textbf{v.a. binomiale} de paramètres
$(n,p)$, s'exprimant à partir de la fonction indicatrice par $X = I_{A_1} + I_{A_2} + \ldots +
I_{A_n}$. Son domaine est $S_X = \{0,1,\ldots,n\}$, et sa loi de probabilité est
\[
p_k = P(X=k) = C_n^k \, p^k (1-p)^{n-k}, \qquad C_n^k = \binom{n}{k} =
\frac{n!}{k!(n-k)!}.
\]
\[
\mu_X = \E[X] = np, \qquad \sigma_X^2 = np(1-p), \qquad G_X(z) = (1-p+pz)^n.
\]
\end{defnbox}

\subsection{Variable aléatoire géométrique}

\begin{defnbox}[title=V.a. géométrique -- nombre d'échecs avant le premier succès]
On compte le nombre d'essais de Bernoulli avant d'obtenir un succès, les expériences de
Bernoulli, de paramètre $p$, étant indépendantes. La \textbf{v.a. géométrique} de paramètre
$p$, $X$, compte le nombre d'\emph{échecs} avant le premier succès. Son domaine est $S_X =
\{0,1,2,\ldots\}$ et sa loi de probabilité est
\[
p_k = P(X=k) = p(1-p)^k.
\]
\[
\mu_X = \E[X] = \frac{1-p}{p}, \qquad \sigma_X^2 = \frac{1-p}{p^2}, \qquad G_X(z) =
\frac{p}{1-(1-p)z}.
\]
\end{defnbox}

\begin{defnbox}[title=V.a. géométrique -- nombre d'essais avant le premier succès]
Dans la seconde version, la v.a. géométrique de paramètre $p$, $X'$, compte le nombre
d'\emph{essais} avant le premier succès. Par conséquent, $X' = X+1$, son domaine est $S_{X'}
= \{1,2,\ldots\}$ et sa loi de probabilité est
\[
p_k' = P(X'=k) = p(1-p)^{k-1}.
\]
\[
\mu_{X'} = \E[X'] = \frac{1}{p}, \qquad \sigma_{X'}^2 = \frac{1-p}{p^2}, \qquad G_{X'}(z) =
\frac{pz}{1-(1-p)z}.
\]
\end{defnbox}

\subsection{Variable aléatoire de Poisson}

\begin{defnbox}[title=V.a. de Poisson]
Dans le processus de Poisson, une v.a. compte le nombre d'arrivées de clients dans un
système pendant une unité de temps. La \textbf{v.a. de Poisson}, de paramètre $\mu > 0$, est
définie par la loi de probabilité
\[
p_k = P(X=k) = \frac{\mu^k}{k!} e^{-\mu}.
\]
\[
\mu_X = \E[X] = \mu, \qquad \sigma_X^2 = \mu, \qquad G_X(z) = e^{\mu(z-1)}.
\]
\end{defnbox}

\begin{propbox}[title=La Poisson comme limite de la binomiale]
La v.a. de Poisson a la propriété d'être la limite d'une v.a. binomiale $B(n,p)$, lorsque
$n \to \infty$ et $p \to 0$ tout en gardant $np$ fini. Précisément, soit $X$ une binomiale
$B(n,p)$ et $Y$ une v.a. de Poisson de paramètre $\mu$. Alors si $n \to \infty$ et $p \to 0$
avec $\mu = np$ fini,
\[
p_k = P(X=k) = C_n^k \, p^k (1-p)^{n-k} \;\longrightarrow\; \frac{\mu^k}{k!} e^{-\mu} =
P(Y=k).
\]
\end{propbox}

\subsection{Variable aléatoire uniforme}

\begin{defnbox}[title=V.a. uniforme continue]
Une \textbf{v.a. uniforme continue} sur un intervalle $[a,b] = S_X$ est définie par sa
densité de probabilité
\[
f_X(x) = \begin{cases} 1/(b-a) & \text{si } a \leq x \leq b \\ 0 & \text{sinon.}
\end{cases}
\]
\[
\mu_X = \E[X] = \frac{a+b}{2}, \qquad \sigma_X^2 = \frac{(b-a)^2}{12}, \qquad \Phi_X(\omega) =
\frac{e^{j\omega b} - e^{j\omega a}}{(b-a)j\omega}.
\]
\end{defnbox}

\subsection{Variable aléatoire gaussienne}

\begin{rembox}[title=Pourquoi la gaussienne est-elle si importante ?]
La gaussienne est une excellente approximation d'une somme d'un grand nombre de v.a. dont la
densité de probabilité n'est pas entièrement connue, ce qui est très souvent le cas en
pratique.
\end{rembox}

\begin{defnbox}[title=V.a. gaussienne]
Son domaine est $S_X = \mathbb{R}$. Une \textbf{v.a. gaussienne} $X$ de paramètres
$(\mu,\sigma)$ est définie par sa densité de probabilité
\[
f_X(x) = \frac{1}{\sigma\sqrt{2\pi}} \, e^{-\frac{(x-\mu)^2}{2\sigma^2}}.
\]
$\mu$ est la moyenne, car $f_X(x)$ est symétrique autour de $x = \mu$ ($\mu_X = \E[X] =
\mu$), et $\sigma$ est l'écart-type ($\sigma_X^2 = \sigma^2$). Sa fonction caractéristique
est
\[
\Phi_X(\omega) = e^{j\mu\omega - \sigma^2\omega^2/2}.
\]
\end{defnbox}

\begin{propbox}[title=Fonction de répartition]
La fonction de répartition doit être calculée numériquement ou à l'aide de tables (fonction
$Q(\cdot)$ ou $\operatorname{erf}(\cdot)$) :
\[
F_X(x) = \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{\frac{x-\mu}{\sigma}} e^{-\xi^2/2}\,d\xi = 1 -
\operatorname{erf}\!\left(\frac{x-\mu}{\sigma}\right)
\]
car cette intégrale n'est pas connue sous forme analytique.
\end{propbox}

\begin{defnbox}[title=V.a. normale]
Une v.a. gaussienne centrée et normalisée, i.e. dont la moyenne $\mu = 0$ et l'écart-type
$\sigma = 1$, est appelée \textbf{v.a. normale}.
\end{defnbox}

\subsection{Variable aléatoire exponentielle}

\begin{defnbox}[title=V.a. exponentielle]
Une \textbf{v.a. exponentielle} mesure le temps entre deux arrivées successives dans un
processus poissonnien. Elle est à valeurs positives $S_X = [0,\infty[$ et paramétrée par un
réel positif $\lambda > 0$. Sa densité de probabilité est
\[
f_X(x) = \lambda e^{-\lambda x}.
\]
\[
F_X(x) = 1 - e^{-\lambda x}, \qquad \mu_X = \frac{1}{\lambda}, \qquad \sigma_X^2 =
\frac{1}{\lambda^2}, \qquad \Phi_X(\omega) = \frac{\lambda}{\lambda - j\omega}.
\]
\end{defnbox}

\begin{propbox}[title=Absence de mémoire]
La v.a. exponentielle est la seule v.a. continue sans mémoire :
\[
P(X > t + T \mid X > t) = P(X > T).
\]
\end{propbox}

\subsection{Variable aléatoire Gamma}

\begin{defnbox}[title=Fonction Gamma]
Pour $u > 0$,
\[
\Gamma(u) = \int_0^{\infty} x^{u-1} e^{-x}\,dx.
\]
Elle possède les propriétés
\[
\Gamma(1/2) = \sqrt{\pi}, \qquad \Gamma(u+1) = u\,\Gamma(u), \qquad \Gamma(u) = (u-1)! \;\;
\text{si } u \in \mathbb{N}.
\]
\end{defnbox}

\begin{defnbox}[title=V.a. Gamma]
Une \textbf{v.a. Gamma} $G(\lambda, \alpha)$ est une v.a. à valeurs strictement positives
($S_X = \,]0,\infty[$), paramétrée par deux réels positifs $\alpha, \lambda > 0$ et définie
par sa densité de probabilité
\[
f_X(x) = \frac{\lambda(\lambda x)^{\alpha-1} e^{-\lambda x}}{\Gamma(\alpha)}.
\]
\[
\mu_X = \frac{\alpha}{\lambda}, \qquad \sigma_X^2 = \frac{\alpha}{\lambda^2}, \qquad
\Phi_X(\omega) = \frac{1}{(1 - j\omega/\lambda)^\alpha}.
\]
\end{defnbox}

\begin{rembox}[title=Cas particuliers de la v.a. Gamma]
Beaucoup de v.a. continues sont des cas particuliers de la v.a. Gamma$(\lambda, \alpha)$.
\begin{itemize}
  \item La v.a. $G(\lambda, m)$ où $m \in \mathbb{N}$ est une \textbf{v.a. d'Erlang} de
        paramètres $(\lambda, m)$. Comme $m$ est entier,
        \[
        f_X(x) = \frac{\lambda(\lambda x)^{m-1} e^{-\lambda x}}{(m-1)!}.
        \]
        En particulier, la v.a. d'Erlang lorsque $m=1$ est une v.a. exponentielle.
  \item Si $\lambda = 1/2$ et $\alpha = \nu/2$ avec $\nu \in \mathbb{N}$, la v.a.
        $G(1/2, \nu/2)$ est une \textbf{v.a. Chi-carré} $\chi^2(\nu)$ à $\nu$ degrés de
        liberté. Dans ce cas,
        \[
        f_X(x) = \frac{x^{(\nu-2)/2} e^{-x/2}}{2^{\nu/2} \Gamma(\nu/2)}.
        \]
\end{itemize}
\end{rembox}
