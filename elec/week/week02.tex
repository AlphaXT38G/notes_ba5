\section{C'est quoi un signal ?}


\begin{definition}

\begin{itemize}

\item Il véhicule une information (le signal véhicule un message).

\item Un phénomène est à l'origine de sa génération (de l'énergie est nécessaire pour le produire : cours d'électrotechnique).

\item Notion dynamique : $I$ (charges en déplacement).

\item Notion statique : $U$ (répartition de charges).

\end{itemize}

\end{definition}


%=========================================================
\section{Loi d'Ohm et puissance électrique}


\begin{formula}

Loi d'Ohm :

\[
U = R\,I
\]

Puissance électrique :

\[
P = U\,I
\]

où :

\begin{itemize}

\item $U$ : tension aux bornes du composant [V] (volt) ;

\item $R$ : résistance [$\Omega$] (ohm) ;

\item $I$ : courant qui traverse le composant [A] (ampère) ;

\item $P$ : puissance électrique [W] (watt), $1\,\text{W} = 1\,\text{V}\cdot\text{A}$.

\end{itemize}

\end{formula}


%=========================================================
\section{Définitions des valeurs d'un signal}


\begin{definition}

\textbf{Valeur crête}

La taille maximale d'une demi-période (on ne compte pas l'offset s'il y en a un).

\textbf{Valeur crête-crête} (ou crête à creux)

De tout en haut à tout en bas du signal.

\textbf{Valeur moyenne}

C'est la valeur moyenne du signal (voir section \textit{Valeurs moyennes}).

\textbf{Valeur efficace}

Voir plus loin (section \textit{Calcul de la valeur efficace}).

\end{definition}

\section{Sources d'énergie}


\begin{center}

\includegraphics[
width=1.0\linewidth
]{../images/week02_01.png}

\end{center}


\begin{definition}

\textbf{Source de tension} (idéale)

Impose une tension $U_0$ [V] entre ses deux bornes ($+$ et $-$), quel que soit le courant qu'elle débite. Le courant dépend du circuit branché dessus. Sa résistance interne est nulle.

\textbf{Source de courant} (idéale)

Impose un courant $i$ [A] dans sa branche (sens de la flèche), quelle que soit la tension à ses bornes. La tension dépend du circuit branché dessus. Sa résistance interne est infinie.

\end{definition}


%=========================================================
\section{Premières conventions}


\begin{important}

\begin{itemize}

\item Signaux continus représentés avec des majuscules : $U$, $V$, $I$.

\item Signaux variables représentés avec des minuscules : $u(t)$, $v(t)$, $i(t)$, $u$, $v$, $i$.

\end{itemize}

\end{important}


\begin{warning}

Le courant d'électrons va dans le sens inverse de la convention du sens du courant électrique, qui est de $+$ vers $-$.

\end{warning}


%=========================================================
\section{Tension et potentiel}


\begin{warning}

Différence de potentiel = tension, mais tension $\neq$ potentiel (le potentiel se calcule toujours en un point de l'espace).

\end{warning}


\begin{definition}

\textbf{Potentiel électrique}

Grandeur définie en \textbf{un point} $A$ du circuit, notée $V_A$ [V]. Elle correspond à l'énergie potentielle électrique par unité de charge en ce point, et se mesure toujours par rapport à un point de référence dont le potentiel vaut $0\,\text{V}$ (la masse).

\end{definition}


\begin{formula}

\[
U_{AB} = V_A - V_B
\]

où :

\begin{itemize}

\item $U_{AB}$ : tension (différence de potentiel) entre les points $A$ et $B$ [V] ;

\item $V_A$ : potentiel du point $A$ [V] ;

\item $V_B$ : potentiel du point $B$ [V].

\end{itemize}

\end{formula}


%=========================================================
\section{Équipotentiel}


\begin{definition}

Le même potentiel partout, par exemple sur tous les segments d'un même fil.

\end{definition}


%=========================================================
\section{Terre et masse}


\begin{important}

\begin{itemize}

\item \textbf{Terre} = référence \textbf{absolue}.

\item \textbf{Masse} = référence \textbf{relative}.

\end{itemize}

\end{important}


\begin{example}

\textbf{Masse :} dans un circuit alimenté par une pile de $9\,\text{V}$, on choisit la borne $-$ comme masse : $V_- = 0\,\text{V}$ et $V_+ = 9\,\text{V}$. Si on choisit plutôt la borne $+$ comme masse : $V_+ = 0\,\text{V}$ et $V_- = -9\,\text{V}$. Les potentiels changent, mais la tension $U = V_+ - V_- = 9\,\text{V}$ reste la même.

\vspace{3pt}

\textbf{Terre :} la prise de terre d'une installation (fil jaune-vert des prises) est reliée physiquement au sol ; c'est la même référence pour tous les appareils.

\end{example}

\section{Valeurs moyennes (domaines discrets)}


\begin{center}

\includegraphics[
width=1.0\linewidth

\begin{formula}

\[
\text{SOMME\_TOTALE} = \sum_i \text{VALEUR}_i
\]

\[
\text{VALEUR\_MOYENNE} = \frac{\text{SOMME\_TOTALE}}{\text{NB\_ECHANTILLONS}}
\]

où :

\begin{itemize}

\item $\text{VALEUR}_i$ : valeur du $i$-ème échantillon (unité du signal, p.\,ex. [V] ou [A]) ;

\item $\text{SOMME\_TOTALE}$ : somme de tous les échantillons (même unité) ;

\item $\text{NB\_ECHANTILLONS}$ : nombre d'échantillons [sans unité] ;

\item $\text{VALEUR\_MOYENNE}$ : valeur moyenne des échantillons (même unité que le signal).

\end{itemize}

\end{formula}


%=========================================================
\section{Valeurs moyennes (domaines continus)}


\textbf{1) Approximation de la surface de la courbe par un histogramme}

Comparable au domaine discret. L'erreur est grande car $\Delta t_i$ est grand.

\begin{center}

\includegraphics[
width=1.0\linewidth
]{../images/week02_03.png}

\end{center}


\textbf{2) Le \og pas \fg{} de calcul est plus fin}

Toujours comparable au domaine discret. L'erreur est petite car $\Delta t_i$ est petit.

\begin{center}

\includegraphics[
width=1.0\linewidth
]{../images/week02_04.png}

\end{center}


\begin{formula}

\[
S_i = H_i \cdot \Delta t_i
\qquad
\text{Surface totale} = \sum_i H_i \cdot \Delta t_i
\]

\[
\text{Moyenne} = \frac{\text{Surface totale}}{\text{NB échantillons} \cdot \Delta t} = \frac{\text{Surface totale}}{\text{Temps total}}
\]

\[
\text{Somme totale} = \sum_i H_i
\qquad
\text{Moyenne} = \frac{\text{Somme totale}}{\text{NB échantillons}}
\]

où :

\begin{itemize}

\item $H_i$ : hauteur de la $i$-ème barre = valeur du signal sur cet intervalle (p.\,ex. [V]) ;

\item $\Delta t_i$ (ou $\Delta t$) : largeur d'une barre = pas de calcul [s] ;

\item $S_i$ : surface d'une barre (p.\,ex. [V$\cdot$s]) ;

\item $\text{Temps total} = \text{NB échantillons} \cdot \Delta t$ : durée analysée [s] ;

\item $\text{Moyenne}$ : valeur moyenne du signal (même unité que $H_i$).

\end{itemize}

\end{formula}


\textbf{3) Le \og pas \fg{} de calcul est infinitésimal : domaine continu}

L'erreur est nulle.

\begin{center}

\includegraphics[
width=1.0\linewidth
]{../images/week02_05.png}

\end{center}


\begin{formula}

\[
dS = H(t)\cdot dt
\qquad
\text{Surface totale} = \int_0^T H(t)\,dt
\]

\[
\text{Moyenne} = \frac{\text{Surface totale}}{\text{Période d'analyse}} = \frac{1}{T}\int_0^T H(t)\,dt
\]

où :

\begin{itemize}

\item $H(t)$ : valeur du signal à l'instant $t$ (p.\,ex. [V]) ;

\item $dt$ : largeur infinitésimale d'une barre [s] ;

\item $dS$ : surface d'une barre infinitésimale (p.\,ex. [V$\cdot$s]) ;

\item $T$ : période d'analyse [s] ;

\item $\text{Moyenne}$ : valeur moyenne du signal (même unité que $H(t)$).

\end{itemize}

\end{formula}

\section{Calcul de la valeur efficace}


\begin{center}

\includegraphics[
width=1.0\linewidth
]{../images/week02_06.png}

\end{center}


\begin{definition}

\textbf{Valeur efficace} (liée à la notion de puissance)

La valeur efficace $U_{EFF}$ d'un signal variable est la tension continue qui dissiperait, dans la même résistance $R$, la même puissance moyenne que ce signal. $P_{EFF}$ est donc comparable à la puissance moyenne.

\end{definition}


\begin{important}

\textbf{Pourquoi pas simplement la valeur moyenne ?}

Une tension alternative (sinusoïde sans offset) est positive la moitié du temps et négative l'autre moitié : sa valeur moyenne vaut $0\,\text{V}$. Pourtant, une ampoule branchée dessus chauffe et éclaire. La moyenne ne dit donc rien de l'énergie transmise.

\vspace{3pt}

\textbf{L'idée}

La puissance dans une résistance vaut $P = U^2/R$ : elle dépend de $U^2$, qui est toujours positif. Le signe de la tension ne compte pas : $+5\,\text{V}$ et $-5\,\text{V}$ chauffent autant. On cherche donc la tension \textbf{continue} $U_{EFF}$ qui chaufferait la résistance exactement autant que le signal variable, en moyenne sur une période.

\vspace{3pt}

\textbf{La formule générale (RMS)}

En anglais : \textit{Root Mean Square} = racine de la moyenne du carré. On égale la puissance moyenne du signal et celle de la tension continue :

\[
\frac{1}{T}\int_0^T \frac{u^2(t)}{R}\,dt = \frac{U_{EFF}^2}{R}
\quad\Rightarrow\quad
U_{EFF} = \sqrt{\frac{1}{T}\int_0^T u^2(t)\,dt}
\]

($R$ se simplifie : la valeur efficace ne dépend que du signal.)

\vspace{3pt}

\textbf{Comment la calculer, en 3 étapes}

\begin{enumerate}[leftmargin=*,itemsep=1pt,topsep=1pt]

\item \textbf{Carré} : on élève le signal au carré, $u^2(t)$ (tout devient positif).

\item \textbf{Moyenne} : on fait la moyenne de $u^2(t)$ sur une période, $\frac{1}{T}\int_0^T u^2(t)\,dt$ (comme dans la section \textit{Valeurs moyennes}, mais sur le carré).

\item \textbf{Racine} : on prend la racine carrée pour revenir en volts.

\end{enumerate}

Pour la sinusoïde $u(t) = A\sin(2\pi f t)$ : le carré vaut en moyenne $A^2/2$ (voir l'exemple ci-dessous), donc $U_{EFF} = \sqrt{A^2/2} = A/\sqrt{2} \approx 0.707\,A$.

\vspace{3pt}

\textbf{Exemple concret}

Les $230\,\text{V}$ d'une prise sont une valeur \textbf{efficace} : la tension crête vaut $A = 230\cdot\sqrt{2} \approx 325\,\text{V}$. Un radiateur branché sur la prise chauffe autant qu'avec $230\,\text{V}$ continus.

\end{important}


\begin{formula}

Énergie = somme de toutes les puissances sur la période :

\[
E = \int_0^T P(t)\,dt = \int_0^T U(t)\,I(t)\,dt
\]

\[
\phantom{E} = \int_0^T \frac{U^2(t)}{R}\,dt = \int_0^T R\,I^2(t)\,dt
\]

(les deux dernières formes si elle est appliquée à une résistance $R$, avec $U = RI$).

\[
P(t) = U(t)\,I(t)
\qquad
P_{EFF} = U_{EFF}\,I_{EFF} = \frac{U_{EFF}^2}{R} = R\,I_{EFF}^2
\]

où :

\begin{itemize}

\item $E$ : énergie dissipée pendant la période [J] (joule), $1\,\text{J} = 1\,\text{W}\cdot\text{s}$ ;

\item $P(t)$ : puissance instantanée [W] ;

\item $U(t)$, $I(t)$ : tension [V] et courant [A] instantanés ;

\item $R$ : résistance [$\Omega$] ;

\item $T$ : période du signal [s], $T = 1/f$ ;

\item $P_{EFF}$ : puissance efficace [W] ;

\item $U_{EFF}$, $I_{EFF}$ : tension efficace [V] et courant efficace [A].

\end{itemize}

\end{formula}


\begin{example}

\textbf{Sinusoïde} $U(t) = A\sin(2\pi f t)$ appliquée à $R$, avec $A$ : amplitude (valeur crête) [V] et $f$ : fréquence [Hz].

\vspace{3pt}

\textbf{1.} On remplace $U(t)$ dans l'énergie, et on linéarise avec $\sin^2 x = \frac{1-\cos 2x}{2}$ :

\[
E = \int_0^T \frac{A^2\sin^2(2\pi f t)}{R}\,dt = \int_0^T \frac{A^2\left[1-\cos(4\pi f t)\right]}{2R}\,dt
\]

\textbf{2.} On intègre (une primitive de $\cos(4\pi f t)$ est $\frac{\sin(4\pi f t)}{4\pi f}$) :

\[
E = \frac{A^2}{R}\left[\frac{t}{2} - \frac{\sin(4\pi f t)}{8\pi f}\right]_0^T
\]

\textbf{3.} On évalue entre $0$ et $T$ : comme $fT = 1$, $\sin(4\pi f T) = \sin(4\pi) = 0$ et $\sin(0) = 0$ :

\[
E = \frac{A^2}{R}\left[\frac{T}{2} - \frac{\sin 4\pi}{8\pi f} - \frac{0}{2} + \frac{\sin(0)}{8\pi f}\right] = \frac{A^2 T}{2R}
\]

\textbf{4.} On divise par $T$ pour obtenir la puissance moyenne, et on l'égale à $U_{EFF}^2/R$ :

\[
\boxed{P_{EFF} = \frac{E}{T} = \frac{A^2}{2R} = \frac{U_{EFF}^2}{R}
\quad\Rightarrow\quad
U_{EFF} = \frac{A}{\sqrt{2}}}
\]

\textbf{A.N.} si $A = 1\,\text{V}$ $\Rightarrow$ $U_{EFF} = \frac{1}{\sqrt 2} \approx 0.707\,\text{V}$.

\end{example}
